	query_sql "UPDATE `solo_stock_player` SET shares = shares + 1, total_cost = total_cost + 100, pending_div = 0, drip_carryover = 0 WHERE account_id = 999999 AND ticker = 'TST1'";

	AssertEquals(1100, .@tc1, "T2: TST1 DRIP total cost updated (1000 + 100 = 1100)");
	AssertEquals(0, .@carry1, "T2: TST1 DRIP carryover is 0");

	.@payout_frac = 250;

	.@new_carry = .@payout_frac % .@price_frac;
	query_sql "UPDATE `solo_stock_player` SET shares = shares + " + .@new_shares + ", total_cost = total_cost + " + (.@new_shares * .@price_frac) + ", drip_carryover = " + .@new_carry + " WHERE account_id = 999999 AND ticker = 'TST1'";

	AssertEquals(13, .@shares1, "T2: TST1 fractional DRIP shares added (11 + 2 = 13)");
	AssertEquals(1300, .@tc1, "T2: TST1 fractional DRIP total cost updated (1100 + 200 = 1300)");
	AssertEquals(50, .@carry1, "T2: TST1 fractional DRIP carryover preserved (50z)");
	AssertEquals(250, (.@new_shares * .@price_frac) + .@new_carry, "T2: DRIP value conservation invariant ((shares * price) + carry == payout)");

	// ==========================================================

		100,  100, 1,   100,  10100,  // Buy 100 @ 100z: subtotal 10,000z, fee 100z -> 10,100z

		  1,   80, 1,     0,     80,  // Buy 1 @ 80z: subtotal 80z, fee 0z (<100z) -> 80z
		100,  100, 0,   100,   9900,  // Sell 100 @ 100z: subtotal 10,000z, fee 100z -> 9,900z
		  1,   80, 0,     0,     80;  // Sell 1 @ 80z: subtotal 80z, fee 0z -> 80z

	.@num_fee_cases = getarraysize(.@fee_cases) / .@fee_stride;
	for (.@i = 0; .@i < .@num_fee_cases; .@i++) {
		.@foff = .@i * .@fee_stride;
		.@f_qty      = .@fee_cases[.@foff];
		.@f_price    = .@fee_cases[.@foff + 1];
		.@f_is_buy   = .@fee_cases[.@foff + 2];
		.@f_exp_fee  = .@fee_cases[.@foff + 3];
		.@f_exp_tot  = .@fee_cases[.@foff + 4];

		.@f_subtotal = .@f_qty * .@f_price;
		.@f_act_fee  = .@f_subtotal / 100;
		.@f_act_tot  = .@f_is_buy ? (.@f_subtotal + .@f_act_fee) : (.@f_subtotal - .@f_act_fee);

		AssertEquals(.@f_exp_fee, .@f_act_fee, "T5: Brokerage fee Case #" + .@i + " (" + .@f_qty + " @ " + .@f_price + "z)");
		AssertEquals(.@f_exp_tot, .@f_act_tot, "T5: Total transaction amount Case #" + .@i);

	// 6. T6: Partial Sale Average Cost Basis & Dividend Pro-rating
	// ==========================================================
	// Setup account 999998: 10 shares of TST2 @ 2000z cost basis (200z/sh), 200z pending dividend
	query_sql "REPLACE INTO `solo_stock_player` (account_id, ticker, shares, total_cost, pending_div, drip_enabled, drip_carryover) VALUES (999998, 'TST2', 10, 2000, 200, 0, 0)";

	.@p_orig_tc = 2000;
	.@p_orig_div = 200;
	.@p_cur_price = 250;
	.@p_tax_rate = 10;

	.@p_subtotal = .@p_cur_price * .@p_sell_qty; // 1000z
	.@p_fee = .@p_subtotal / 100; // 10z
	.@p_net_proceeds = .@p_subtotal - .@p_fee; // 990z

	// Pro-rate pending dividend on partial sale
	.@p_div_gross = (.@p_orig_div * .@p_sell_qty) / .@p_orig_shares; // 80z
	.@p_div_tax = (.@p_div_gross * .@p_tax_rate) / 100; // 8z
	.@p_div_net = .@p_div_gross - .@p_div_tax; // 72z
	.@p_total_received = .@p_net_proceeds + .@p_div_net; // 1062z

	AssertEquals(80, .@p_div_gross, "T6: Pro-rated gross dividend on 4 shares is 80z (4/10 of 200z)");
	AssertEquals(72, .@p_div_net, "T6: Pro-rated net dividend after 10% tax is 72z");
	AssertEquals(1062, .@p_total_received, "T6: Total received on partial liquidation (990z net + 72z div = 1062z)");

	AssertEquals(1200, .@rem_tc, "T6: Remaining cost basis in DB (2000 - 800 = 1200z)");
	AssertEquals(120, .@rem_div, "T6: Remaining pending dividend in DB (200 - 80 = 120z)");
	AssertEquals(200, .@rem_tc / .@rem_shares, "T6: Conserved Average Cost Basis per share invariant (1200z / 6 = 200z/sh)");

	// ==========================================================

	// Matrix: [price, qty, expected_valid] (Stride = 3)
	setarray .@oflow_cases[0],
		  100, 1000000, 1,  // 100z * 1,000,000 = 100,000,000z <= 2B (Valid)

		 5000,  400001, 0,  // 5,000z * 400,001 = 2,000,005,000z > 2B (Rejected)
		50000,   40000, 1,  // 50,000z * 40,000 = 2,000,000,000z == 2B (Valid limit)
		50000,   40001, 0;  // 50,000z * 40,001 = 2,000,050,000z > 2B (Rejected)
