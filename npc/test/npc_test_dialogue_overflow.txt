// COMPLIANT: Paginated dynamic loop (5 items per page with next;)
-	script	test_dialogue_loop_paginated	-1,{
OnInit:
	end;

OnTalk:
	.@total_items = 12;
	.@pages = (.@total_items + 4) / 5;
	for (.@p = 0; .@p < .@pages; .@p++) {
		clear;
		mes "[Inventory - Page " + (.@p + 1) + "/" + .@pages + "]";
		.@start = .@p * 5;
		.@end = .@start + 5;
		if (.@end > .@total_items) .@end = .@total_items;
		for (.@i = .@start; .@i < .@end; .@i++) {
			mes "- Item #" + (.@i + 1);
		}
		next;
	}
	close;
}
