-	script	junk_trader_test#ci	-1,{
OnInit:
	debugmes "Junk Trader CI Test: Starting intensive test suite...";

	// 1. T1: Trigger Daily Rotation Controller
	donpcevent "JunkTraderV2Controller::OnRotateDaily";

	.@roster_day = $JunkRosterDay;
	.@expected_day = gettime(DT_DAYOFYEAR);
	AssertEquals(.@expected_day, .@roster_day, "T1: JunkTraderV2Controller updated $JunkRosterDay");

	.@roster_size = $JunkRosterSize;
	AssertEquals(5, .@roster_size, "T1: $JunkRosterSize defaults to 5 from solo_server_config");

	// 2. T2: Verify 6-Tier Roster Population & Price Bounds
	.@total_valid = 0;
	for (.@tier = 1; .@tier <= 6; .@tier++) {
		for (.@idx = 1; .@idx <= 5; .@idx++) {
			.@item_id = getd("$JunkT" + .@tier + "_" + .@idx);
			.@price = getd("$JunkPriceT" + .@tier + "_" + .@idx);

			if (.@item_id > 0) {
				.@min = 0; .@max = 0;
				query_sql("SELECT min_price, max_price FROM `custom_junk_pool` WHERE tier = " + .@tier + " AND item_id = " + .@item_id, .@min, .@max);

				if (.@min > 0 && .@max > 0) {
					.@in_bounds = (.@price >= .@min && .@price <= .@max);
					AssertTrue(.@in_bounds, "T2: Tier " + .@tier + " Item " + .@idx + " price (" + .@price + "z) within [" + .@min + "z, " + .@max + "z]");
					.@total_valid++;
				}
			}
		}
	}
	AssertEquals(30, .@total_valid, "T2: All 30 daily bounty slots (6 tiers * 5 items) successfully populated with valid price ranges");

	// 3. T3: Verify SQL Mapreg Join with custom_junk_pool (Matching Web Bounty Tab)
	deletearray .@web_item_ids[0], getarraysize(.@web_item_ids);
	deletearray .@web_prices[0], getarraysize(.@web_prices);
	deletearray .@web_item_names$[0], getarraysize(.@web_item_names$);
	deletearray .@web_mob_names$[0], getarraysize(.@web_mob_names$);
	deletearray .@web_mob_lvs[0], getarraysize(.@web_mob_lvs);

	.@web_count = query_sql("SELECT CAST(m1.value AS UNSIGNED), CAST(m2.value AS UNSIGNED), COALESCE(c.item_name, ''), COALESCE(c.mob_name, ''), COALESCE(c.mob_lv, 0) FROM mapreg m1 JOIN mapreg m2 ON m2.varname = REPLACE(m1.varname, '$JunkT', '$JunkPriceT') LEFT JOIN custom_junk_pool c ON c.item_id = CAST(m1.value AS UNSIGNED) AND c.tier = CAST(SUBSTRING(m1.varname, 7, 1) AS UNSIGNED) WHERE m1.varname LIKE '$JunkT%' ORDER BY m1.varname ASC", .@web_item_ids, .@web_prices, .@web_item_names$, .@web_mob_names$, .@web_mob_lvs);

	AssertEquals(30, .@web_count, "T3: Web extraction SQL query returned 30 active bounties from mapreg join");

	// Verify first item has valid item name, mob name and mob level
	.@has_valid_meta = (getstrlen(.@web_item_names$[0]) > 0 && getstrlen(.@web_mob_names$[0]) > 0 && .@web_mob_lvs[0] > 0);
	AssertTrue(.@has_valid_meta, "T3: Web bounty items join correctly with monster metadata (" + .@web_item_names$[0] + " from " + .@web_mob_names$[0] + " Lv." + .@web_mob_lvs[0] + ")");

	// 4. T4: Daily Quota & Payout Arithmetic Simulation
	.@daily_limit = 100;
	.@test_sold = 80;
	.@test_price = 5000;
	.@attempt_qty = 30; // 80 + 30 = 110 (exceeds 100 quota)

	.@remaining_quota = .@daily_limit - .@test_sold;
	AssertEquals(20, .@remaining_quota, "T4: Remaining daily quota calculated accurately (100 - 80 = 20)");

	// Attempting to sell 30 when remaining is 20 must be clamped to 20
	.@actual_sell = min(.@attempt_qty, .@remaining_quota);
	AssertEquals(20, .@actual_sell, "T4: Sale quantity clamped to remaining quota");

	.@payout = .@test_price * .@actual_sell;
	AssertEquals(100000, .@payout, "T4: Payout calculation (20 * 5000z = 100,000z)");

	debugmes "Junk Trader CI Test: Completed successfully.";
	end;
}
