// --- Global Market Logic ---
-	script	StockExchangeSetup	-1,{
OnInit:
	// Ponytail: Market simulation is handled by Elysia background service.
	// Only ensure base market metadata exists in DB on boot.
	query_sql "INSERT IGNORE INTO `solo_stock_meta` (mkey, mval) VALUES ('MarketMood', 1), ('MarketDrift', 0), ('LastUpdate', UNIX_TIMESTAMP()), ('DivTaxRate', 10), ('BlackSwanChance', 2), ('LatestEventTime', 0)";
	end;
}

function	script	F_StockExchangeMain	{
L_MenuMain:
	.@account_id = getcharid(3);

	// Ponytail: Query stock tickers and metadata dynamically from solo_stock_market (active enabled only)
	deletearray .@tickers$[0], getarraysize(.@tickers$);
	deletearray .@names$[0], getarraysize(.@names$);
	deletearray .@prices[0], getarraysize(.@prices);
	deletearray .@price_olds[0], getarraysize(.@price_olds);
	deletearray .@divs[0], getarraysize(.@divs);

	.@stock_count = query_sql("SELECT ticker, name, price, price_old, dividend FROM `solo_stock_market` WHERE enabled = 1 ORDER BY ticker ASC", .@tickers$, .@names$, .@prices, .@price_olds, .@divs);

	mes "[Midgard Stock Exchange]";
	if (.@stock_count <= 0) {
		mes "The stock market is currently closed for maintenance.";
		close;
	}

	.@total_pages = (.@stock_count + 4) / 5;
	for (.@page = 0; .@page < .@total_pages; .@page++) {
		clear;
		mes "Standard Index [Page " + (.@page + 1) + "/" + .@total_pages + "]:";
		mes "---------------------------";
		.@start_idx = .@page * 5;
		.@end_idx = .@start_idx + 5;
		if (.@end_idx > .@stock_count) .@end_idx = .@stock_count;
		for (.@i = .@start_idx; .@i < .@end_idx; .@i++) {
			.@p = .@prices[.@i];
			.@po = .@price_olds[.@i];
			.@diff = .@p - .@po;
			if (.@diff > 0) { .@clr$ = "^00FF00"; .@d$ = "(+" + .@diff + "z)"; }
			else if (.@diff < 0) { .@clr$ = "^FF0000"; .@d$ = "(" + .@diff + "z)"; }
			else { .@clr$ = "^000000"; .@d$ = "(0z)"; }
			mes "- " + .@tickers$[.@i] + " (" + .@names$[.@i] + "): " + .@clr$ + .@p + "z^000000 " + .@d$ + " [Div: " + .@divs[.@i] + "z]";
		}
		next;
	}

	if (getgmlevel() >= 99) {
		switch(select("Trade Buy:Trade Sell:My Portfolio:Claim Dividends:DRIP Settings:^FF0000[Admin Tools]^000000:Exit")) {
			case 1: goto L_MenuBuy; case 2: goto L_MenuSell; case 3: goto L_MenuPort; case 4: goto L_ClaimDiv; case 5: goto L_MenuDrip; case 6: goto L_Admin; default: return;
		}
	} else {
		switch(select("Trade Buy:Trade Sell:My Portfolio:Claim Dividends:DRIP Settings:Exit")) {
			case 1: goto L_MenuBuy; case 2: goto L_MenuSell; case 3: goto L_MenuPort; case 4: goto L_ClaimDiv; case 5: goto L_MenuDrip; default: return;
		}
	}

L_ClaimDiv:
	.@account_id = getcharid(3);
	.@total = 0;
	query_sql "SELECT COALESCE(SUM(pending_div), 0) FROM `solo_stock_player` WHERE account_id = " + .@account_id, .@total;
	if (.@total <= 0) {
		mes "[Midgard Stock Exchange]";
		mes "No dividends to collect.";
		next; goto L_MenuMain;
	}

	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@tax_rate;
	
	.@tax_amt = (.@total * .@tax_rate) / 100;
	.@net = .@total - .@tax_amt;

	// MAX_ZENY overflow protection (MAX_ZENY = 2,147,483,647)
	.@zeny_space = 2147483647 - Zeny;
	if (.@zeny_space <= 0) {
		mes "[Midgard Stock Exchange]";
		mes "^FF0000Your wallet is full! You cannot hold any more Zeny.^000000";
		mes "Please deposit or spend your Zeny before claiming dividends.";
		next; goto L_MenuMain;
	}

	if (.@net > .@zeny_space) {
		.@payout = .@zeny_space;
		.@gross_reduced = (.@total * .@payout) / .@net;
		query_sql "UPDATE `solo_stock_player` SET pending_div = pending_div - " + .@gross_reduced + " WHERE account_id = " + .@account_id;
		.@net = .@payout;
		mes "[Midgard Stock Exchange]";
		mes "^FF0000Warning: Partial collection due to wallet limit. Remainder stays in your dividend balance.^000000";
	} else {
		query_sql "UPDATE `solo_stock_player` SET pending_div = 0 WHERE account_id = " + .@account_id;
	}
	
	mes "[Dividend Distribution]";
	mes "Gross Dividends: ^00FF00" + .@total + "z^000000";
	mes "Tax (" + .@tax_rate + "%): ^FF0000-" + .@tax_amt + "z^000000";
	mes "---------------------------";
	mes "Net Collected: ^0000FF" + .@net + "z^000000";
	
	Zeny = Zeny + .@net;
	callfunc "F_RecordEconomy", 2, .@net; // 2: Total Zeny Earned
	next; goto L_MenuMain;

L_Admin:
	mes "[GM Admin Console]";
	switch(select("Set Dividend Tax:Reset My Portfolio:Reset Index:Back")) {
		case 1:
			query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@current_tax;
			mes "Current Dividend Tax: " + .@current_tax + "%";
			mes "Input new tax rate (0-100):";
			input .@new_tax;
			if (.@new_tax < 0 || .@new_tax > 100) { mes "Invalid rate."; next; goto L_Admin; }
			query_sql "UPDATE `solo_stock_meta` SET mval = " + .@new_tax + " WHERE mkey = 'DivTaxRate'";
			mes "Dividend Tax updated to " + .@new_tax + "%.";
			next; goto L_Admin;
		case 2:
			.@account_id = getcharid(3);
			query_sql "DELETE FROM `solo_stock_player` WHERE account_id = " + .@account_id;
			mes "Portfolio wiped."; next; goto L_MenuMain;
		case 3: 
			query_sql "UPDATE `solo_stock_market` SET price = 100, price_old = 100, dividend = 3, div_acc = 0, split_count = 0";
			mes "Market Index reset."; next; goto L_MenuMain;
		default: goto L_MenuMain;
	}

L_MenuBuy:
	mes "[Buy Stocks]";
	if (.@stock_count <= 0) {
		mes "No stock listings available.";
		next; goto L_MenuMain;
	}

	.@buy_menu$ = "";
	for (.@i = 0; .@i < .@stock_count; .@i++) {
		if (.@i > 0) .@buy_menu$ = .@buy_menu$ + ":";
		.@buy_menu$ = .@buy_menu$ + .@tickers$[.@i] + " - " + .@names$[.@i] + " (" + .@prices[.@i] + "z)";
	}
	.@buy_menu$ = .@buy_menu$ + ":Cancel";

	.@choice = select(.@buy_menu$);
	if (.@choice <= 0 || .@choice > .@stock_count) goto L_MenuMain;

	.@sel_idx = .@choice - 1;
	.@cn$ = .@tickers$[.@sel_idx];
	.@account_id = getcharid(3);
	
	.@p = 0;
	query_sql "SELECT price FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p;
	if (.@p <= 0) {
		mes "Error: Stock price unavailable.";
		next; goto L_MenuMain;
	}
	
	mes "Selected: ^0000FF" + .@cn$ + " (" + .@names$[.@sel_idx] + ")^000000";
	mes "Price: " + .@p + "z";
	mes "A 1.0% brokerage transaction fee will be added to the purchase.";
	mes "Enter quantity to buy (Max 1,000,000):";
	input .@qty;
	if (.@qty <= 0 || .@qty > 1000000) {
		mes "Invalid quantity. Must be between 1 and 1,000,000.";
		next; goto L_MenuMain;
	}

	// Guard against 32-bit integer overflow
	if (.@qty > (2000000000 / .@p)) {
		mes "Order value exceeds single transaction limit.";
		next; goto L_MenuMain;
	}

	.@subtotal = .@p * .@qty;
	.@fee = .@subtotal / 100;
	.@total = .@subtotal + .@fee;
	if (.@total <= 0) {
		mes "Arithmetic calculation error.";
		next; goto L_MenuMain;
	}

	mes "Subtotal: " + .@subtotal + "z";
	mes "Broker Fee (1%): " + .@fee + "z";
	mes "Total Cost: " + .@total + "z";
	next;
	if (select("Purchase:Cancel") == 2) goto L_MenuMain;
	if (Zeny < .@total) { mes "Insufficient Zeny."; next; goto L_MenuMain; }
	Zeny = Zeny - .@total; 
	
	query_sql "INSERT INTO `solo_stock_player` (account_id, ticker, shares, total_cost) VALUES (" + .@account_id + ", '" + .@cn$ + "', " + .@qty + ", " + .@subtotal + ") ON DUPLICATE KEY UPDATE shares = shares + " + .@qty + ", total_cost = total_cost + " + .@subtotal;
	goto L_MenuMain;

L_MenuSell:
	mes "[Sell Stocks]";
	.@account_id = getcharid(3);
	
	deletearray .@held_tickers$[0], getarraysize(.@held_tickers$);
	deletearray .@held_shares[0], getarraysize(.@held_shares);
	deletearray .@held_costs[0], getarraysize(.@held_costs);
	deletearray .@held_divs[0], getarraysize(.@held_divs);

	.@held_count = query_sql("SELECT ticker, shares, total_cost, pending_div FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND shares > 0 ORDER BY ticker ASC", .@held_tickers$, .@held_shares, .@held_costs, .@held_divs);
	if (.@held_count <= 0) {
		mes "You do not currently own any shares.";
		next; goto L_MenuMain;
	}

	.@sell_menu$ = "";
	for (.@i = 0; .@i < .@held_count; .@i++) {
		if (.@i > 0) .@sell_menu$ = .@sell_menu$ + ":";
		.@sell_menu$ = .@sell_menu$ + .@held_tickers$[.@i] + " (" + .@held_shares[.@i] + " shares)";
	}
	.@sell_menu$ = .@sell_menu$ + ":Cancel";

	.@choice = select(.@sell_menu$);
	if (.@choice <= 0 || .@choice > .@held_count) goto L_MenuMain;

	.@sel_idx = .@choice - 1;
	.@cn$ = .@held_tickers$[.@sel_idx];
	.@s = .@held_shares[.@sel_idx];
	.@tc = .@held_costs[.@sel_idx];
	.@pending = .@held_divs[.@sel_idx];

	.@p = 0;
	query_sql "SELECT price FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p;
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@tax_rate;
	
	if (.@s <= 0 || .@p <= 0) { mes "No valid shares."; next; goto L_MenuMain; }
	
	switch(select("Amount:Liquidate All:Back")) {
		case 1: mes "Amount to sell (Max " + .@s + "):"; input .@qty; break;
		case 2: .@qty = .@s; break;
		default: goto L_MenuMain;
	}
	if (.@qty <= 0 || .@qty > .@s) goto L_MenuMain;

	// Guard against 32-bit overflow
	if (.@qty > (2000000000 / .@p)) {
		mes "Order exceeds transaction limit. Please sell in smaller batches.";
		next; goto L_MenuMain;
	}
	
	.@subtotal = .@p * .@qty;
	.@fee = .@subtotal / 100;
	.@net = .@subtotal - .@fee;

	// Pro-rate pending dividends on sale
	.@payout_gross = (.@pending * .@qty) / .@s;
	.@payout_tax = (.@payout_gross * .@tax_rate) / 100;
	.@payout_net = .@payout_gross - .@payout_tax;
	
	mes "Subtotal (" + .@qty + " shares @ " + .@p + "z): " + .@subtotal + "z";
	mes "Broker Fee (1%): -" + .@fee + "z";
	mes "Net Stock Proceeds: ^00FF00" + .@net + "z^000000";
	if (.@payout_gross > 0) {
		mes "Pro-rated Dividends (Net): ^0000FF+" + .@payout_net + "z^000000 (Gross " + .@payout_gross + "z | Tax -" + .@payout_tax + "z)";
	}
	mes "---------------------------";
	.@total_payout = .@net + .@payout_net;
	mes "Total Received: ^00FF00" + .@total_payout + "z^000000";
	next;
	if (select("Confirm:Cancel") == 2) goto L_MenuMain;

	// Check MAX_ZENY wallet capacity
	.@zeny_space = 2147483647 - Zeny;
	if (.@total_payout > .@zeny_space) {
		mes "[Midgard Stock Exchange]";
		mes "^FF0000Warning: Your wallet cannot hold the full payout.^000000";
		mes "Available space: " + .@zeny_space + "z.";
		mes "Please deposit or spend some Zeny before selling this quantity.";
		next; goto L_MenuMain;
	}
	
	if (.@qty == .@s) {
		query_sql "UPDATE `solo_stock_player` SET shares = 0, total_cost = 0, pending_div = 0 WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
	} else {
		query_sql "UPDATE `solo_stock_player` SET shares = shares - " + .@qty + ", total_cost = total_cost - ROUND((total_cost * " + .@qty + ") / shares), pending_div = pending_div - " + .@payout_gross + " WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
	}
	Zeny = Zeny + .@total_payout;
	callfunc "F_RecordEconomy", 2, .@total_payout; // 2: Total Zeny Earned
	goto L_MenuMain;

L_MenuPort:
	.@account_id = getcharid(3);
	deletearray .@port_tickers$[0], getarraysize(.@port_tickers$);
	deletearray .@port_shares[0], getarraysize(.@port_shares);
	deletearray .@port_costs[0], getarraysize(.@port_costs);
	deletearray .@port_pending[0], getarraysize(.@port_pending);

	.@port_count = query_sql("SELECT ticker, shares, total_cost, pending_div FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND shares > 0 ORDER BY ticker ASC", .@port_tickers$, .@port_shares, .@port_costs, .@port_pending);
	if (.@port_count <= 0) {
		mes "[My Portfolio]";
		mes "Your portfolio is currently empty.";
		next; goto L_MenuMain;
	}

	.@port_per_page = 2;
	.@port_pages = (.@port_count + .@port_per_page - 1) / .@port_per_page;
	for (.@p = 0; .@p < .@port_pages; .@p++) {
		clear;
		mes "[My Portfolio - Page " + (.@p + 1) + "/" + .@port_pages + "]";
		mes "---------------------------";
		.@start_p = .@p * .@port_per_page;
		.@end_p = .@start_p + .@port_per_page;
		if (.@end_p > .@port_count) .@end_p = .@port_count;
		for (.@i = .@start_p; .@i < .@end_p; .@i++) {
			.@cn$ = .@port_tickers$[.@i];
			.@s = .@port_shares[.@i];
			.@tc = .@port_costs[.@i];
			.@pending = .@port_pending[.@i];
			.@price = 0; .@div = 0; .@sname$ = "";
			query_sql "SELECT name, price, dividend FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@sname$, .@price, .@div;

			.@avg = .@tc / .@s;
			.@pl = .@price - .@avg;
			if (.@pl > 0) .@pl$ = "^00FF00+" + .@pl + "z^000000";
			else if (.@pl < 0) .@pl$ = "^FF0000" + .@pl + "z^000000";
			else .@pl$ = "0z";
			
			if (.@avg > 0) { .@yoc_p = (.@div * 1000) / .@avg; .@yoc$ = (.@yoc_p/10) + "." + (.@yoc_p % 10) + "%"; } else .@yoc$ = "0.0%";
			if (.@price > 0) { .@cur_p = (.@div * 1000) / .@price; .@cur$ = (.@cur_p/10) + "." + (.@cur_p % 10) + "%"; } else .@cur$ = "0.0%";
			
			mes "- " + .@cn$ + " (" + .@sname$ + "): " + .@s + " shares";
			mes "  Avg: " + .@avg + "z | Market: " + .@price + "z | P/L: " + .@pl$;
			mes "  ^777777Yield: ^000000" + .@cur$ + " | ^0000FFYoC: " + .@yoc$ + "^000000 | Pending: ^0000FF" + .@pending + "z^000000";
		}
		next;
	}
	goto L_MenuMain;

L_MenuDrip:
	.@account_id = getcharid(3);
	mes "[DRIP Settings]";
	mes "Dividend Reinvestment Plan (DRIP) allows you to automatically reinvest your dividends to buy more shares, tax-free!";
	mes "----------------------------------";
	
	deletearray .@drip_tickers$[0], getarraysize(.@drip_tickers$);
	deletearray .@drip_shares[0], getarraysize(.@drip_shares);
	deletearray .@drip_enabled[0], getarraysize(.@drip_enabled);
	deletearray .@drip_carry[0], getarraysize(.@drip_carry);

	.@drip_count = query_sql("SELECT ticker, shares, drip_enabled, drip_carryover FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND shares > 0 ORDER BY ticker ASC", .@drip_tickers$, .@drip_shares, .@drip_enabled, .@drip_carry);
	if (.@drip_count <= 0) {
		mes "You do not currently hold any stocks.";
		next; goto L_MenuMain;
	}
	
	for (.@i = 0; .@i < .@drip_count; .@i++) {
		mes "- " + .@drip_tickers$[.@i] + ": " + (.@drip_enabled[.@i] ? "^00FF00ON^000000" : "^FF0000OFF^000000") + " | Carryover: " + .@drip_carry[.@i] + "z (Holdings: " + .@drip_shares[.@i] + " shares)";
	}
	next;
	
	mes "Select a ticker to toggle DRIP:";
	.@menu_str$ = "";
	for (.@i = 0; .@i < .@drip_count; .@i++) {
		if (.@i > 0) .@menu_str$ = .@menu_str$ + ":";
		.@menu_str$ = .@menu_str$ + .@drip_tickers$[.@i] + " (" + (.@drip_enabled[.@i] ? "^00FF00DRIP ON^000000" : "^FF0000DRIP OFF^000000") + ")";
	}
	.@menu_str$ = .@menu_str$ + ":Back";
	
	.@choice = select(.@menu_str$);
	if (.@choice > .@drip_count || .@choice <= 0) goto L_MenuMain;
	
	.@cn$ = .@drip_tickers$[.@choice - 1];
	.@de = .@drip_enabled[.@choice - 1];
	.@new_de = .@de ? 0 : 1;
	
	query_sql "UPDATE `solo_stock_player` SET drip_enabled = " + .@new_de + " WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
	mes "DRIP for " + .@cn$ + " is now " + (.@new_de ? "^00FF00ON^000000" : "^FF0000OFF^000000") + ".";
	next; goto L_MenuDrip;
}
