// --- Global Market Logic ---
-	script	StockExchangeSetup	-1,{
OnInit:
	setarray $StockCities$[0],"PRT","GEF","MOR","PAY","ALB";
	
	// Ensure Tables Exist
	query_sql "CREATE TABLE IF NOT EXISTS `solo_stock_market` (`ticker` VARCHAR(10) PRIMARY KEY, `price` INT DEFAULT 100, `price_old` INT DEFAULT 100, `dividend` INT DEFAULT 3, `div_acc` INT DEFAULT 0, `split_count` INT DEFAULT 0) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4";
	query_sql "CREATE TABLE IF NOT EXISTS `solo_stock_player` (`account_id` INT(11) UNSIGNED NOT NULL, `ticker` VARCHAR(10) NOT NULL, `shares` INT DEFAULT 0, `total_cost` BIGINT DEFAULT 0, `pending_div` INT DEFAULT 0, `drip_enabled` TINYINT DEFAULT 0, `drip_carryover` INT DEFAULT 0, PRIMARY KEY (`account_id`, `ticker`)) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4";
	
	// Check and Add DRIP Columns
	query_sql "SHOW COLUMNS FROM `solo_stock_player` LIKE 'drip_enabled'", .@col$;
	if (.@col$ == "") {
		query_sql "ALTER TABLE `solo_stock_player` ADD `drip_enabled` TINYINT DEFAULT 0";
	}
	query_sql "SHOW COLUMNS FROM `solo_stock_player` LIKE 'drip_carryover'", .@col$;
	if (.@col$ == "") {
		query_sql "ALTER TABLE `solo_stock_player` ADD `drip_carryover` INT DEFAULT 0";
	}
	query_sql "SHOW COLUMNS FROM `solo_stock_player` LIKE 'pending_div'", .@col$;
	if (.@col$ == "") {
		query_sql "ALTER TABLE `solo_stock_player` ADD `pending_div` INT DEFAULT 0";
		// Perform one-time migration for existing pending dividends
		query_sql "UPDATE `solo_stock_player` p JOIN `solo_stock_market` m ON p.ticker = m.ticker SET p.pending_div = p.shares * (m.div_acc - p.last_claim_acc) WHERE p.shares > 0";
	}

	query_sql "CREATE TABLE IF NOT EXISTS `solo_stock_meta` (`mkey` VARCHAR(32) PRIMARY KEY, `mval` INT DEFAULT 0) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4";

	// Initialize Market Data
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		query_sql "INSERT IGNORE INTO `solo_stock_market` (ticker) VALUES ('" + $StockCities$[.@i] + "')";
	}
	query_sql "INSERT IGNORE INTO `solo_stock_meta` (mkey, mval) VALUES ('MarketMood', 1), ('MarketDrift', 0), ('LastUpdate', 0), ('DivTaxRate', 10)";

	// Catch-up logic for missed midnights
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'LastUpdate'", $StockLastUpdate;
	set .@now, gettimetick(2);

	if ($StockLastUpdate <= 31) {
		// Migration from old format (day of month) or fresh install
		set $StockLastUpdate, .@now;
		query_sql "UPDATE `solo_stock_meta` SET mval = " + $StockLastUpdate + " WHERE mkey = 'LastUpdate'";
	} else {
		// Calculate missed midnights using DATEDIFF
		query_sql "SELECT DATEDIFF(CURDATE(), DATE(FROM_UNIXTIME(" + $StockLastUpdate + ")))", .@days_passed;
		
		if (.@days_passed > 0) {
			if (.@days_passed > 30) set .@days_passed, 30; // Hard cap
			
			setarray .@names$[0],"PRT","GEF","MOR","PAY","ALB";
			for (set .@d, 0; .@d < .@days_passed; set .@d, .@d + 1) {
				for (set .@i, 0; .@i < 5; set .@i, .@i + 1) { 
					query_sql "UPDATE `solo_stock_market` SET price_old = price WHERE ticker = '" + .@names$[.@i] + "'";
				}
				set $MarketDrift, rand(-4, 6);
				query_sql "UPDATE `solo_stock_meta` SET mval = " + $MarketDrift + " WHERE mkey = 'MarketDrift'";
				callsub L_MidnightProc;
			}
			debugmes "Stock Market: Caught up on " + .@days_passed + " missed days.";
			
			set $StockLastUpdate, .@now;
			query_sql "UPDATE `solo_stock_meta` SET mval = " + $StockLastUpdate + " WHERE mkey = 'LastUpdate'";
		}
	}
	end;

OnClock0000:
	setarray .@names$[0],"PRT","GEF","MOR","PAY","ALB";
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) { 
		query_sql "UPDATE `solo_stock_market` SET price_old = price WHERE ticker = '" + .@names$[.@i] + "'";
	}
	set $MarketDrift, rand(-4, 6);
	query_sql "UPDATE `solo_stock_meta` SET mval = " + $MarketDrift + " WHERE mkey = 'MarketDrift'";
	callsub L_MidnightProc;
	set $StockLastUpdate, gettimetick(2);
	query_sql "UPDATE `solo_stock_meta` SET mval = " + $StockLastUpdate + " WHERE mkey = 'LastUpdate'";
OnClock0300:
OnClock0600:
OnClock0900:
OnClock1200:
OnClock1500:
OnClock1800:
OnClock2100:
	callsub L_ShiftProc;
	end;

OnShiftManual: 
	callsub L_ShiftProc;
	end;

OnMidnightManual: 
	setarray .@names$[0],"PRT","GEF","MOR","PAY","ALB";
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) { 
		query_sql "UPDATE `solo_stock_market` SET price_old = price WHERE ticker = '" + .@names$[.@i] + "'";
	}
	set $MarketDrift, rand(-4, 6);
	query_sql "UPDATE `solo_stock_meta` SET mval = " + $MarketDrift + " WHERE mkey = 'MarketDrift'";
	callsub L_MidnightProc;
	set $StockLastUpdate, gettimetick(2);
	query_sql "UPDATE `solo_stock_meta` SET mval = " + $StockLastUpdate + " WHERE mkey = 'LastUpdate'";
	announce "MARKET NEWS: Admin has triggered a Midnight Cycle. Dividends updated.", bc_all;
	end;

L_ShiftProc:
	setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
	set .@u, 0;
	// Sync Meta from SQL
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'MarketMood'", $MarketMood;
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'MarketDrift'", $MarketDrift;

	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@c$[.@i];
		query_sql "SELECT price, dividend, split_count FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p, .@div, .@sc;
		
		set .@f, $MarketDrift + rand(-10, 10);
		if ($MarketMood == 1) set .@f, .@f + 3;
		else if ($MarketMood == 2) set .@f, .@f - 3;
		else if ($MarketMood == 3) set .@f, .@f + rand(-20, 30);
		
		set .@new, .@p + (.@p * .@f / 100);
		if (.@new < 50) set .@new, 50; 
		if (.@new >= 1000) {
			set .@new, .@new / 10;
			query_sql "UPDATE `solo_stock_market` SET price = " + .@new + ", price_old = price_old / 10, dividend = dividend / 10, split_count = split_count + 1 WHERE ticker = '" + .@cn$ + "'";
			query_sql "UPDATE `solo_stock_player` SET shares = shares * 10 WHERE ticker = '" + .@cn$ + "'";
			announce "MARKET NEWS: Successful 10:1 Stock Split for "+.@cn$+"!", bc_all;
		} else {
			query_sql "UPDATE `solo_stock_market` SET price = " + .@new + " WHERE ticker = '" + .@cn$ + "'";
		}
		if (.@new > .@p) set .@u, .@u + 1;
	}
	if (rand(1, 100) <= 8) set $MarketMood, 3; 
	else if (.@u >= 2) set $MarketMood, 1; 
	else set $MarketMood, 2;
	
	query_sql "UPDATE `solo_stock_meta` SET mval = " + $MarketMood + " WHERE mkey = 'MarketMood'";
	announce "MARKET NEWS: Stock shift complete. Trend: " + callfunc("F_MoodNameStr") + ".", bc_all;
	return;

L_MidnightProc:
	setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'MarketMood'", $MarketMood;
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@c$[.@i];
		query_sql "SELECT price, dividend, div_acc FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p, .@div, .@divacc;
		
		set .@target, (.@p * 25) / 1000;
		if (.@target < 1) set .@target, 1;
		
		if (rand(1, 100) <= 30) {
			if (.@div < .@target && $MarketMood != 2) { 
				set .@div, .@div + 1; 
				announce "CORPORATE NEWS: "+.@cn$+" dividend hike (+1z)!", bc_all; 
			}
			else if (.@div > .@target && $MarketMood != 1) { 
				set .@div, .@div - 1; 
				announce "CORPORATE NEWS: "+.@cn$+" dividend cut (-1z).", bc_all; 
			}
		}
		
		if ($MarketMood == 3) { set .@div, 0; announce "MARKET CRISIS: "+.@cn$+" has suspended its dividend!", bc_all; }
		else if (.@div <= 0) { set .@div, 1; announce "CORPORATE NEWS: "+.@cn$+" has restored its dividend.", bc_all; }
		
		if (.@div > 500) set .@div, 500;
		query_sql "UPDATE `solo_stock_market` SET dividend = " + .@div + ", div_acc = div_acc + " + .@div + " WHERE ticker = '" + .@cn$ + "'";
		query_sql "UPDATE `solo_stock_player` SET pending_div = pending_div + (shares * " + .@div + ") WHERE ticker = '" + .@cn$ + "' AND shares > 0";

		// DRIP Reinvestment Processing
		deletearray .@u_accs; deletearray .@u_shares; deletearray .@u_costs; deletearray .@u_carrys; deletearray .@u_pendings;
		query_sql "SELECT account_id, shares, total_cost, drip_carryover, pending_div FROM `solo_stock_player` WHERE ticker = '" + .@cn$ + "' AND drip_enabled = 1 AND shares > 0", .@u_accs, .@u_shares, .@u_costs, .@u_carrys, .@u_pendings;
		set .@num_users, getarraysize(.@u_accs);
		for (set .@k, 0; .@k < .@num_users; set .@k, .@k + 1) {
			set .@p_acc, .@u_accs[.@k];
			set .@p_sh, .@u_shares[.@k];
			set .@p_tc, .@u_costs[.@k];
			set .@p_carry, .@u_carrys[.@k];
			set .@p_pending, .@u_pendings[.@k];
			
			set .@payout, .@p_pending + .@p_carry;
			if (.@payout >= .@p) {
				set .@new_shares, .@payout / .@p;
				set .@cost, .@new_shares * .@p;
				set .@new_carry, .@payout % .@p;
				query_sql "UPDATE `solo_stock_player` SET shares = shares + " + .@new_shares + ", total_cost = total_cost + " + .@cost + ", pending_div = 0, drip_carryover = " + .@new_carry + " WHERE account_id = " + .@p_acc + " AND ticker = '" + .@cn$ + "'";
			} else {
				query_sql "UPDATE `solo_stock_player` SET pending_div = 0, drip_carryover = " + .@payout + " WHERE account_id = " + .@p_acc + " AND ticker = '" + .@cn$ + "'";
			}
		}
	}
	return;
}

function	script	F_StockExchangeMain	{
L_MenuMain:
	.@account_id = getcharid(3);
	setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";

	mes "[Midgard Stock Exchange]";
	clear;
	mes "Standard Index (Trends Today):";
	mes "---------------------------";
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@c$[.@i];
		query_sql "SELECT price, price_old, dividend FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p, .@po, .@div;
		set .@diff, .@p - .@po;
		if (.@diff > 0) { set .@clr$, "^00FF00"; set .@d$, "(+"+.@diff+"z)"; }
		else if (.@diff < 0) { set .@clr$, "^FF0000"; set .@d$, "("+.@diff+"z)"; }
		else { set .@clr$, "^000000"; set .@d$, "(0z)"; }
		mes "- " + .@cn$ + ": " + .@clr$ + .@p + "z^000000 " + .@d$ + " [Div: "+.@div+"z]";
	}
	next;

	if (getgmlevel() >= 99) {
		switch(select("Trade Buy:Trade Sell:My Portfolio:Claim Dividends:DRIP Settings:^FF0000[Admin Tools]^000000:Exit")) {
			case 1: goto L_MenuBuy; case 2: goto L_MenuSell; case 3: goto L_MenuPort; case 4: goto L_ClaimDiv; case 5: goto L_MenuDrip; case 6: goto L_Admin; default: return;
		}
	} else {
		switch(select("Trade Buy:Trade Sell:My Portfolio:Claim Dividends:DRIP Settings:Exit")) {
			case 1: goto L_MenuBuy; case 2: goto L_MenuSell; case 3: goto L_MenuPort; case 4: goto L_ClaimDiv; case 5: goto L_MenuDrip; default: return;
		}
	}

L_ClaimDiv:
	set .@total, 0; setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
	.@account_id = getcharid(3);
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@tax_rate;

	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@c$[.@i];
		set .@pending, 0;
		query_sql "SELECT pending_div FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'", .@pending;
		if (.@pending > 0) {
			set .@total, .@total + .@pending;
			query_sql "UPDATE `solo_stock_player` SET pending_div = 0 WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
		}
	}
	if (.@total <= 0) { mes "No dividends to collect."; next; goto L_MenuMain; }
	
	set .@tax_amt, (.@total * .@tax_rate) / 100;
	set .@net, .@total - .@tax_amt;
	
	mes "Gross Dividends: ^00FF00" + .@total + "z^000000";
	mes "Tax (" + .@tax_rate + "%): ^FF0000-" + .@tax_amt + "z^000000";
	mes "---------------------------";
	mes "Net Collected: ^0000FF" + .@net + "z^000000";
	
	set Zeny, Zeny + .@net;
	callfunc "F_RecordEconomy", 2, .@net; // 2: Total Zeny Earned
	next; goto L_MenuMain;

L_Admin:
	mes "[GM Admin Console]";
	switch(select("Force 3-Hr Price Shift:Force Midnight Cycle:Set Dividend Tax:Reset My Portfolio:Reset Index:Back")) {
		case 1: doevent "StockExchangeSetup::OnShiftManual"; close;
		case 2: doevent "StockExchangeSetup::OnMidnightManual"; close;
		case 3:
			query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@current_tax;
			mes "Current Dividend Tax: " + .@current_tax + "%";
			mes "Input new tax rate (0-100):";
			input .@new_tax;
			if (.@new_tax < 0 || .@new_tax > 100) { mes "Invalid rate."; next; goto L_Admin; }
			query_sql "UPDATE `solo_stock_meta` SET mval = " + .@new_tax + " WHERE mkey = 'DivTaxRate'";
			mes "Dividend Tax updated to " + .@new_tax + "%.";
			next; goto L_Admin;
		case 4:
			.@account_id = getcharid(3);
			query_sql "DELETE FROM `solo_stock_player` WHERE account_id = " + .@account_id;
			mes "Portfolio wiped."; next; goto L_MenuMain;
		case 5: 
			query_sql "UPDATE `solo_stock_market` SET price = 100, price_old = 100, dividend = 3, div_acc = 0, split_count = 0";
			mes "Market Index reset."; next; goto L_MenuMain;
		default: goto L_MenuMain;
	}

L_MenuBuy:
	mes "[Buy Stocks]";
	set .@city, select("PRT:GEF:MOR:PAY:ALB");
	setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
	set .@cn$, .@c$[.@city - 1];
	.@account_id = getcharid(3);
	
	set .@p, 0;
	query_sql "SELECT price FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p;
	
	mes "Price: " + .@p + "z";
	mes "A 1.0% brokerage transaction fee will be added to the purchase.";
	input .@qty;
	if (.@qty <= 0) goto L_MenuMain;
	set .@subtotal, .@p * .@qty;
	set .@fee, .@subtotal / 100;
	set .@total, .@subtotal + .@fee;
	mes "Subtotal: " + .@subtotal + "z";
	mes "Broker Fee (1%): " + .@fee + "z";
	mes "Total Cost: " + .@total + "z";
	next;
	if (select("Purchase:Cancel") == 2) goto L_MenuMain;
	if (Zeny < .@total) { mes "No Zeny."; next; goto L_MenuMain; }
	set Zeny, Zeny - .@total; 
	
	query_sql "INSERT INTO `solo_stock_player` (account_id, ticker, shares, total_cost) VALUES (" + .@account_id + ", '" + .@cn$ + "', " + .@qty + ", " + .@subtotal + ") ON DUPLICATE KEY UPDATE shares = shares + " + .@qty + ", total_cost = total_cost + " + .@subtotal;
	goto L_MenuMain;

L_MenuSell:
	mes "[Sell Stocks]";
	setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
	.@account_id = getcharid(3);
	
	deletearray .@sh;
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@q_shares, 0;
		query_sql "SELECT shares FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND ticker = '" + .@c$[.@i] + "'", .@q_shares;
		set .@sh[.@i], .@q_shares;
	}
	
	set .@city, select("PRT("+.@sh[0]+"):GEF("+.@sh[1]+"):MOR("+.@sh[2]+"):PAY("+.@sh[3]+"):ALB("+.@sh[4]+")");
	set .@cn$, .@c$[.@city - 1];
	set .@p, 0;
	set .@s, 0;
	set .@tc, 0;
	set .@pending, 0;
	query_sql "SELECT price FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p;
	query_sql "SELECT shares, total_cost, pending_div FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'", .@s, .@tc, .@pending;
	query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'DivTaxRate'", .@tax_rate;
	
	if (.@s <= 0) { mes "No shares."; next; goto L_MenuMain; }
	
	set .@payout_gross, .@pending;
	set .@payout_tax, (.@payout_gross * .@tax_rate) / 100;
	set .@payout_net, .@payout_gross - .@payout_tax;
	
	mes "Accrued Dividends (Net): ^00FF00+" + .@payout_net + "z^000000";
	if (.@payout_gross > 0) mes "  ^777777(Gross: " + .@payout_gross + "z | Tax " + .@tax_rate + "%: -" + .@payout_tax + "z)^000000";
	next;
	switch(select("Amount:Liquidate All:Back")) {
		case 1: mes "Amount?"; input .@qty; break;
		case 2: set .@qty, .@s; break;
		default: goto L_MenuMain;
	}
	if (.@qty <= 0 || .@qty > .@s) goto L_MenuMain;
	
	set .@subtotal, .@p * .@qty;
	set .@fee, .@subtotal / 100;
	set .@net, .@subtotal - .@fee;
	
	mes "Subtotal: " + .@subtotal + "z";
	mes "Broker Fee (1%): " + .@fee + "z";
	mes "Net Received: " + .@net + "z";
	next;
	if (select("Confirm:Cancel") == 2) goto L_MenuMain;
	
	set .@redCost, (.@tc * .@qty) / .@s;
	query_sql "UPDATE `solo_stock_player` SET shares = shares - " + .@qty + ", total_cost = total_cost - " + .@redCost + ", pending_div = 0 WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
	set Zeny, Zeny + .@net + .@payout_net;
	callfunc "F_RecordEconomy", 2, .@net + .@payout_net; // 2: Total Zeny Earned
	goto L_MenuMain;

L_MenuPort:
	mes "[My Portfolio]";
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@c$[.@i];
		set .@s, 0;
		set .@tc, 0;
		set .@pending, 0;
		set .@p, 0;
		set .@div, 0;
		query_sql "SELECT shares, total_cost, pending_div FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'", .@s, .@tc, .@pending;
		query_sql "SELECT price, dividend FROM `solo_stock_market` WHERE ticker = '" + .@cn$ + "'", .@p, .@div;

		if (.@s > 0) {
			set .@avg, .@tc / .@s; set .@pl, .@p - .@avg;
			if (.@pl > 0) set .@pl$, "^00FF00+" + .@pl + "z^000000"; else if (.@pl < 0) set .@pl$, "^FF0000" + .@pl + "z^000000"; else set .@pl$, "0z";
			
			if (.@avg > 0) { set .@yoc_p, (.@div * 1000) / .@avg; set .@yoc$, (.@yoc_p/10) + "." + (.@yoc_p % 10) + "%"; } else set .@yoc$, "0.0%";
			if (.@p > 0) { set .@cur_p, (.@div * 1000) / .@p; set .@cur$, (.@cur_p/10) + "." + (.@cur_p % 10) + "%"; } else set .@cur$, "0.0%";
			
			mes "- " + .@cn$ + ": " + .@s + " shares";
			mes "  Avg: " + .@avg + "z | Market: " + .@p + "z | P/L: " + .@pl$;
			mes "  ^777777Yield: ^000000" + .@cur$ + " (Market) | ^0000FFYoC: " + .@yoc$ + " (Average)^000000";
			mes "  ^0000FFPending Div: " + .@pending + "z^000000";
		}
	}
	next;
	goto L_MenuMain;

L_MenuDrip:
	mes "[DRIP Settings]";
	mes "Dividend Reinvestment Plan (DRIP) allows you to automatically reinvest your dividends to buy more shares, tax-free!";
	mes "----------------------------------";
	
	// Query each ticker individually to avoid getarraysize() truncation on trailing zero values
	setarray .@all_tickers$[0],"PRT","GEF","MOR","PAY","ALB";
	set .@num_holdings, 0;
	deletearray .@held_tickers$; deletearray .@held_shares; deletearray .@held_drip; deletearray .@held_carry;
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		set .@cn$, .@all_tickers$[.@i];
		set .@q_shares, 0;
		set .@q_drip, 0;
		set .@q_carry, 0;
		query_sql "SELECT shares, drip_enabled, drip_carryover FROM `solo_stock_player` WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'", .@q_shares, .@q_drip, .@q_carry;
		if (.@q_shares > 0) {
			set .@held_tickers$[.@num_holdings], .@cn$;
			set .@held_shares[.@num_holdings], .@q_shares;
			set .@held_drip[.@num_holdings], .@q_drip;
			set .@held_carry[.@num_holdings], .@q_carry;
			set .@num_holdings, .@num_holdings + 1;
		}
	}
	
	if (.@num_holdings <= 0) {
		mes "You do not currently hold any stocks.";
		next; goto L_MenuMain;
	}
	
	for (set .@i, 0; .@i < .@num_holdings; set .@i, .@i + 1) {
		mes "- " + .@held_tickers$[.@i] + ": " + (.@held_drip[.@i] ? "^00FF00ON^000000" : "^FF0000OFF^000000") + " | Carryover: " + .@held_carry[.@i] + "z (Holdings: " + .@held_shares[.@i] + " shares)";
	}
	next;
	
	mes "Select a ticker to toggle DRIP:";
	set .@menu_str$, "";
	for (set .@i, 0; .@i < .@num_holdings; set .@i, .@i + 1) {
		set .@menu_str$, .@menu_str$ + .@held_tickers$[.@i] + " (" + (.@held_drip[.@i] ? "^00FF00DRIP ON^000000" : "^FF0000DRIP OFF^000000") + "):";
	}
	set .@menu_str$, .@menu_str$ + "Back";
	
	set .@choice, select(.@menu_str$);
	if (.@choice > .@num_holdings || .@choice <= 0) goto L_MenuMain;
	
	set .@cn$, .@held_tickers$[.@choice - 1];
	set .@de, .@held_drip[.@choice - 1];
	set .@new_de, .@de ? 0 : 1;
	
	query_sql "UPDATE `solo_stock_player` SET drip_enabled = " + .@new_de + " WHERE account_id = " + .@account_id + " AND ticker = '" + .@cn$ + "'";
	mes "DRIP for " + .@cn$ + " is now " + (.@new_de ? "^00FF00ON^000000" : "^FF0000OFF^000000") + ".";
	next; goto L_MenuDrip;
}

function	script	F_MoodNameStr	{
	if ($MarketMood == 1) return "Bullish"; if ($MarketMood == 2) return "Bearish"; if ($MarketMood == 3) return "Chaos"; return "Neutral";
}
