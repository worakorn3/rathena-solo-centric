// --- Logic Controller ---
-	script	Adventurer_AI	-1,{
OnTimer10000:
	// Move Prontera Party
	.@x = rand(100, 250);
	.@y = rand(100, 250);
	npcwalkto .@x, .@y, "Knight#adv_prt";
	npcwalkto .@x+rand(-2,2), .@y+rand(-2,2), "Priest#adv_prt";
	npcwalkto .@x+rand(-3,3), .@y+rand(-3,3), "Hunter#adv_prt";

	// Move Payon Party
	.@x = rand(150, 300);
	.@y = rand(150, 300);
	npcwalkto .@x, .@y, "Assassin#adv_pay";
	npcwalkto .@x+rand(-2,2), .@y+rand(-2,2), "Wizard#adv_pay";

	initnpctimer;
	end;

OnInit:
	initnpctimer;
	end;
}
