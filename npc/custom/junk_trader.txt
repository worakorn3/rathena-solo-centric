// --- Global Logic ---
function	script	F_JunkTraderMain	{
	if ($LastRotationDay != gettime(4)) {
		set $LastRotationDay, gettime(4);
		setarray .@pool[0], 909, 912, 916, 919, 920, 921, 922, 923, 924, 925, 926, 927;
		set $TodayJunk1, .@pool[rand(0, 3)];
		set $TodayJunk2, .@pool[rand(4, 7)];
		set $TodayJunk3, .@pool[rand(8, 11)];
		set $JunkPrice1, rand(800, 1500);
		set $JunkPrice2, rand(3500, 5500);
		set $JunkPrice3, rand(9000, 18000);
		announce "MARKET NEWS: Daily Junk Merchants have updated their bounties!", bc_all;
	}

	if (gettime(4) != #LastJunkDay) {
		set #DailyJunkSold, 0;
		set #LastJunkDay, gettime(4);
	}

	mes "[Daily Merchant]";
	mes "Welcome. High-value junk for today:";
	mes "- ^0000FF"+getitemname($TodayJunk1)+"^000000: ^00FF00"+$JunkPrice1+"z^000000";
	mes "- ^0000FF"+getitemname($TodayJunk2)+"^000000: ^00FF00"+$JunkPrice2+"z^000000";
	mes "- ^0000FF"+getitemname($TodayJunk3)+"^000000: ^00FF00"+$JunkPrice3+"z^000000";
	next;
	mes "[Daily Merchant]";
	mes "Your Daily Quota: ^0000FF"+#DailyJunkSold+"/50 Items^000000";
	next;

	if (#DailyJunkSold >= 50) {
		mes "Sorry, I've bought enough today!";
		return;
	}

	switch(select("Sell "+getitemname($TodayJunk1)+":Sell "+getitemname($TodayJunk2)+":Sell "+getitemname($TodayJunk3)+":Cancel")) {
		case 1: set .@id, $TodayJunk1; set .@price, $JunkPrice1; break;
		case 2: set .@id, $TodayJunk2; set .@price, $JunkPrice2; break;
		case 3: set .@id, $TodayJunk3; set .@price, $JunkPrice3; break;
		default: return;
	}

	mes "Selling " + getitemname(.@id) + ". How many?";
	input .@qty;
	if (.@qty <= 0) return;
	if (.@qty > (50 - #DailyJunkSold)) { mes "Exceeds daily sell limit."; return; }
	if (countitem(.@id) < .@qty) { mes "You don't have that many!"; return; }

	delitem .@id, .@qty;
	set Zeny, Zeny + (.@price * .@qty);
	set #DailyJunkSold, #DailyJunkSold + .@qty;
	mes "Transaction complete. Here is " + (.@price * .@qty) + "z.";
	return; 
}
