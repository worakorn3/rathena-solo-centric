prontera,143,172,4	script	Junk Merchant#prt	101,{ callfunc "F_JunkTraderMain"; close; end; }
geffen,120,110,4	script	Junk Merchant#geff	101,{ callfunc "F_JunkTraderMain"; close; end; }
morroc,156,97,4	script	Junk Merchant#morr	101,{ callfunc "F_JunkTraderMain"; close; end; }
payon,152,233,4	script	Junk Merchant#payo	101,{ callfunc "F_JunkTraderMain"; close; end; }
alberta,116,61,4	script	Junk Merchant#albe	101,{ callfunc "F_JunkTraderMain"; close; end; }

// --- Centralized Logic ---
function	script	F_JunkTraderMain	{
	if ($LastRotationDay != gettime(4)) {
		set $LastRotationDay, gettime(4);
		set $TodayJunk1, 909; // Jellopy
		set $TodayJunk2, 912; // Fluff
		set $TodayJunk3, 916; // Feather
		set $JunkPrice1, rand(800, 1500);
		set $JunkPrice2, rand(3500, 5500);
		set $JunkPrice3, rand(9000, 18000);
	}

	if (gettime(4) != #LastJunkDay) {
		set #DailyJunkSold, 0;
		set #LastJunkDay, gettime(4);
	}

	mes "[Daily Junk Merchant]";
	mes "Materials I am buying premium today:";
	mes "- ^0000FF"+getitemname($TodayJunk1)+"^000000: ^00FF00"+$JunkPrice1+"z^000000";
	mes "- ^0000FF"+getitemname($TodayJunk2)+"^000000: ^00FF00"+$JunkPrice2+"z^000000";
	mes "- ^0000FF"+getitemname($TodayJunk3)+"^000000: ^00FF00"+$JunkPrice3+"z^000000";
	next;
	
	mes "[Daily Junk Merchant]";
	mes "Your Quota: ^0000FF"+#DailyJunkSold+"/50 Items Today^000000";
	next;

	if (#DailyJunkSold >= 50) {
		mes "My budget is spent. See you tomorrow!";
		return;
	}

	switch(select("Sell "+getitemname($TodayJunk1)+":Sell "+getitemname($TodayJunk2)+":Sell "+getitemname($TodayJunk3)+":Cancel")) {
		case 1: set .@id, $TodayJunk1; set .@price, $JunkPrice1; break;
		case 2: set .@id, $TodayJunk2; set .@price, $JunkPrice2; break;
		case 3: set .@id, $TodayJunk3; set .@price, $JunkPrice3; break;
		default: return;
	}

	mes "How many " + getitemname(.@id) + " do you have to provide?";
	input .@qty;

	if (.@qty <= 0) return;
	if (.@qty > (50 - #DailyJunkSold)) {
		mes "I can only take " + (50 - #DailyJunkSold) + " more items today.";
		return;
	}

	if (countitem(.@id) < .@qty) {
		mes "You don't have that many " + getitemname(.@id) + "!";
		return;
	}

	delitem .@id, .@qty;
	set Zeny, Zeny + (.@price * .@qty);
	set #DailyJunkSold, #DailyJunkSold + .@qty;
	
	mes "Excellent. Here is your " + (.@price * .@qty) + " Zeny payment.";
	return; 
}
