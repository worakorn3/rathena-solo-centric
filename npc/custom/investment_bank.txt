// ===============================================================
// Solo-Centric Investment Bank
// ===============================================================
// Provides an alternative to Kafra Storage with a deposit fee 
// but daily interest generation. Good for long-term Zeny sinking
// and wealth generation.
//
// Dual-Access Architecture: Mirroring Midgard Stock Exchange
// State is stored in solo_bank_account table, shared with Web Portal.
// ===============================================================

prontera,165,180,4	script	Investment Banker	4_M_BARBER,{
	.@account_id = getcharid(3);
	.@balance = 0;
	.@deposit_time = 0;
	.@interest_paid_total = 0;

	// Load dynamic bank configuration from SQL
	.@interest_rate_bps = 100;
	.@max_days = 10;
	.@deposit_fee_bps = 200;
	.@max_principal = 1900000000;
	.@min_deposit = 100;

	.@cfg_count = query_sql("SELECT `config_key`, `config_value` FROM `solo_bank_config`", .@cfg_keys$, .@cfg_vals);
	for (.@i = 0; .@i < .@cfg_count; .@i++) {
		if (.@cfg_keys$[.@i] == "interest_rate_bps") .@interest_rate_bps = .@cfg_vals[.@i];
		else if (.@cfg_keys$[.@i] == "max_accrual_days") .@max_days = .@cfg_vals[.@i];
		else if (.@cfg_keys$[.@i] == "deposit_fee_bps") .@deposit_fee_bps = .@cfg_vals[.@i];
		else if (.@cfg_keys$[.@i] == "max_principal_limit") .@max_principal = .@cfg_vals[.@i];
		else if (.@cfg_keys$[.@i] == "min_deposit_zeny") .@min_deposit = .@cfg_vals[.@i];
	}

	.@daily_pct = .@interest_rate_bps / 100;
	.@max_pct = (.@interest_rate_bps * .@max_days) / 100;
	.@fee_pct = .@deposit_fee_bps / 100;

	mes "[Midgard Investment Bank]";
	mes "Welcome to the Midgard Sovereign Bank.";
	mes "Deposit fee: ^FF0000" + .@fee_pct + "%^000000";
	mes "Interest rate: ^00FF00" + .@daily_pct + "% per day^000000 (Max " + .@max_pct + "% accumulation)";
	next;
	
	// Read from dedicated table (dual-access parity with Web Portal)
	query_sql "SELECT `principal`, `deposit_time`, `interest_paid_total` FROM `solo_bank_account` WHERE `account_id` = " + .@account_id, .@balance, .@deposit_time, .@interest_paid_total;

	// Check if player has an existing balance and calculate pending interest
	.@days = 0;
	.@interest = 0;
	.@subday_remainder = 0;
	if (.@balance > 0 && .@deposit_time > 0) {
		.@elapsed = gettimetick(2) - .@deposit_time;
		if (.@elapsed > 0) {
			.@days = .@elapsed / 86400; // 86400 seconds = 1 day
			if (.@days > .@max_days) .@days = .@max_days; // Cap interest at max days
			.@interest = (((.@balance * .@interest_rate_bps) / 10000) * .@days);
			if (.@days < .@max_days) {
				.@subday_remainder = .@elapsed % 86400;
			}
		}
	}
	.@total_available = .@balance + .@interest;

	mes "[Investment Bank]";
	mes "Your current principal: ^0000FF" + F_InsertComma(.@balance) + " Zeny^000000";
	mes "Accumulated interest: ^00FF00" + F_InsertComma(.@interest) + " Zeny^000000 (" + .@days + " days)";
	if (.@interest_paid_total > 0) {
		mes "Lifetime interest earned: ^008800" + F_InsertComma(.@interest_paid_total) + " Zeny^000000";
	}
	mes "Total available: ^0000FF" + F_InsertComma(.@total_available) + " Zeny^000000";
	next;

	switch(select("Deposit Zeny:Withdraw Zeny:Cancel")) {
		case 1:
			mes "[Investment Bank]";
			mes "How much would you like to deposit?";
			mes "Example: ^0000FF100m^000000, ^0000FF50k^000000, or exact Zeny.";
			next;
			input .@amount$;
			.@amount = callfunc("F_ParseZenyInput", .@amount$);
			
			if (.@amount < .@min_deposit) {
				mes "[Investment Bank]";
				mes "The minimum deposit is " + F_InsertComma(.@min_deposit) + " Zeny.";
				close;
			}

			if (Zeny < .@amount) {
				mes "[Investment Bank]";
				mes "You don't have that much Zeny.";
				close;
			}

			.@fee = (.@amount * .@deposit_fee_bps) / 10000;
			.@net_deposit = .@amount - .@fee;

			if (.@total_available + .@net_deposit > .@max_principal) {
				mes "[Investment Bank]";
				mes "We cannot accept this deposit. Your principal would exceed our " + F_InsertComma(.@max_principal) + " Zeny limit.";
				close;
			}

			// Deduct from inventory safely
			Zeny -= .@amount;
			
			.@new_principal = .@total_available + .@net_deposit;
			.@now = gettimetick(2);
			.@new_deposit_time = (.@balance > 0 && .@deposit_time > 0) ? (.@now - .@subday_remainder) : .@now;

			// Atomic upsert into dedicated table with compound interest audit
			query_sql "INSERT INTO `solo_bank_account` (`account_id`, `principal`, `deposit_time`, `interest_paid_total`) VALUES (" + .@account_id + ", " + .@new_principal + ", " + .@new_deposit_time + ", " + .@interest + ") ON DUPLICATE KEY UPDATE `principal` = VALUES(`principal`), `deposit_time` = VALUES(`deposit_time`), `interest_paid_total` = `interest_paid_total` + " + .@interest;

			mes "[Investment Bank]";
			mes "You deposited " + F_InsertComma(.@amount) + " Zeny (Fee: " + F_InsertComma(.@fee) + "z).";
			if (.@interest > 0) {
				mes "Compounded Interest: ^00FF00+" + F_InsertComma(.@interest) + " Zeny^000000 added to principal.";
			}
			mes "New Principal: ^0000FF" + F_InsertComma(.@new_principal) + " Zeny^000000.";
			if (.@subday_remainder > 0) {
				.@hours = .@subday_remainder / 3600;
				mes "Preserved Progress: ^00FF00" + .@hours + " hours^000000 toward your next interest day.";
			} else {
				mes "Your interest timer has started.";
			}
			close;

		case 2:
			if (.@total_available <= 0) {
				mes "[Investment Bank]";
				mes "You have no funds to withdraw.";
				close;
			}

			mes "[Investment Bank]";
			mes "How much would you like to withdraw?";
			mes "Available: ^0000FF" + F_InsertComma(.@total_available) + " Zeny^000000";
			next;
			input .@amount$;
			.@amount = callfunc("F_ParseZenyInput", .@amount$);

			if (.@amount < 1) {
				mes "[Investment Bank]";
				mes "Please enter a valid amount to withdraw.";
				close;
			}

			if (.@amount > .@total_available) {
				mes "[Investment Bank]";
				mes "You cannot withdraw more than your available balance of " + F_InsertComma(.@total_available) + " Zeny.";
				close;
			}

			// Prevent Zeny overflow (Max Zeny is 2,100,000,000 in standard RO)
			if (2100000000 - Zeny < .@amount) {
				mes "[Investment Bank]";
				mes "You cannot hold that much Zeny in your inventory.";
				mes "Please make room before withdrawing.";
				close;
			}

			// Give Zeny
			Zeny += .@amount;

			.@new_principal = .@total_available - .@amount;
			.@new_time = (.@new_principal > 0) ? (gettimetick(2) - .@subday_remainder) : 0;
			query_sql "UPDATE `solo_bank_account` SET `principal` = " + .@new_principal + ", `deposit_time` = " + .@new_time + ", `interest_paid_total` = `interest_paid_total` + " + .@interest + " WHERE `account_id` = " + .@account_id;

			mes "[Investment Bank]";
			mes "You have successfully withdrawn ^0000FF" + F_InsertComma(.@amount) + " Zeny^000000.";
			if (.@new_principal > 0) {
				mes "Remaining principal: ^0000FF" + F_InsertComma(.@new_principal) + " Zeny^000000.";
			}
			mes "Thank you for doing business with us.";
			close;

		case 3:
			close;
	}

OnInit:
	waitingroom "Midgard Investment Bank", 0;
	end;
}
