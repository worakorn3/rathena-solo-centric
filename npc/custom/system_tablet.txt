function	script	System_Tablet	{
	doevent "System_Tablet_NPC::OnUse";
	end;
}

-	script	System_Tablet_NPC	-1,{
OnUse:
	mes "[System Tablet]";
	mes "Welcome to the central system.";
	mes "Please select an app to launch:";
	switch(select("Progression Guide:Monster Intel App:Market Pulse App:Dispatch Operations App:Cancel")) {
		case 1:
			mes "[System Tablet]";
			mes "Solo-Centric Global Enhancements:";
			mes "^0000FF1.^000000 Your Storage holds up to 800 slots.";
			mes "^0000FF2.^000000 Card Drops are flat 2x (MvP excluded).";
			mes "^0000FF3.^000000 Kafra Warp fees are slashed by 50%.";
			mes "^0000FF4.^000000 Death penalty is reduced to 0.5%.";
			mes " ";
			mes "Progression Note: Experience rates are ^0000FFtiered^000000:";
			mes "- Lv 1-30: 2x";
			mes "- Lv 31-60: 3x";
			mes "- Lv 61-99: 4x";
			mes "- Lv 100-150: 5x";
			mes "- Lv 151+: 6x";
			close;
		case 2:
			mes "[Monster Intel App]";
			mes "Loading Hunters Guild Database...";
			mes "Please input the exact ^0000FFMonster Name^000000 or ^0000FFID^000000 you wish to research.";
			next;
			input .@mob$;
			
			// Determine if input is ID or Name
			if (atoi(.@mob$) > 0 && getmonsterinfo(atoi(.@mob$), MOB_NAME) != "null") {
				.@mob_id = atoi(.@mob$);
			} else if (getmonsterinfo(.@mob$, MOB_ID) > 0) {
				.@mob_id = getmonsterinfo(.@mob$, MOB_ID);
			} else {
				mes "[Monster Intel App]";
				mes "Error: That monster does not exist in the database.";
				close;
			}
			
			.@mob_lv = getmonsterinfo(.@mob_id, MOB_LV);
			.@price_basic = .@mob_lv * 100;
			.@price_deep = .@mob_lv * 500;
			
			// Fetch Unlock Status (0 = Not Unlocked, 1 = Basic, 2 = Deep)
			.@intel_lv = getd("MobIntel_" + .@mob_id);
			
			mes "[Monster Intel App]";
			mes "Target Acquired: ^0000FF" + getmonsterinfo(.@mob_id, MOB_NAME) + "^000000 (Lv " + .@mob_lv + ")";
			
			if (.@intel_lv == 0) mes "Status: ^FF0000[No Intel Unlocked]^000000";
			else if (.@intel_lv == 1) mes "Status: ^FF8800[Basic Intel Unlocked]^000000";
			else mes "Status: ^00FF00[Deep Intel Unlocked]^000000";
			next;
			
			// Dynamic Menu Array builder
			if (.@intel_lv == 0) {
				.@menu$ = "Purchase Basic Intel ("+.@price_basic+"z):Purchase Direct Full Intel ("+(.@price_basic+.@price_deep)+"z):Cancel";
			} else if (.@intel_lv == 1) {
				.@menu$ = "View Basic Intel:Purchase Deep Intel ("+.@price_deep+"z):Cancel";
			} else {
				.@menu$ = "View Basic Intel:View Deep Intel:Cancel";
			}
			
			switch(select(.@menu$)) {
				case 1:
					if (.@intel_lv == 0) {
						if (Zeny < .@price_basic) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= .@price_basic;
						setd "MobIntel_" + .@mob_id, 1;
						mes "[System Tablet]";
						mes "Basic Intel package permanently unlocked!";
						next;
					}
					mes "[Basic Intel]";
					mes "Name: " + getmonsterinfo(.@mob_id, MOB_NAME);
					mes "HP: " + getmonsterinfo(.@mob_id, MOB_MAXHP);
					mes "Level: " + getmonsterinfo(.@mob_id, MOB_LV);
					mes "Element Lv: " + getmonsterinfo(.@mob_id, MOB_ELEMENTLV) + " (Type " + getmonsterinfo(.@mob_id, MOB_ELEMENT) + ")";
					mes "Size: " + getmonsterinfo(.@mob_id, MOB_SIZE);
					mes "Base EXP: " + getmonsterinfo(.@mob_id, MOB_BASEEXP) + " | Job EXP: " + getmonsterinfo(.@mob_id, MOB_JOBEXP);
					close;
					
				case 2:
					if (.@intel_lv == 0) {
						if (Zeny < (.@price_basic + .@price_deep)) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= (.@price_basic + .@price_deep);
						setd "MobIntel_" + .@mob_id, 2;
						mes "[System Tablet]";
						mes "Deep Intel completely unlocked!";
						next;
					} else if (.@intel_lv == 1) {
						if (Zeny < .@price_deep) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= .@price_deep;
						setd "MobIntel_" + .@mob_id, 2;
						mes "[System Tablet]";
						mes "Upgraded to Deep Intel package!";
						next;
					}
					
					mes "[Deep Intel]";
					mes "ATK: " + getmonsterinfo(.@mob_id, MOB_ATKMIN) + " ~ " + getmonsterinfo(.@mob_id, MOB_ATKMAX);
					mes "DEF: " + getmonsterinfo(.@mob_id, MOB_DEF) + " | MDEF: " + getmonsterinfo(.@mob_id, MOB_MDEF);
					mes "--- Drops ---";
					
					// Clear previous results to avoid leakage
					deletearray $@MobDrop_item[0], getarraysize($@MobDrop_item);
					deletearray $@MobDrop_rate[0], getarraysize($@MobDrop_rate);
					
					if (getmobdrops(.@mob_id)) {
						.@count = getarraysize($@MobDrop_item);
						for (.@i = 0; .@i < .@count; .@i++) {
							mes "- " + getitemname($@MobDrop_item[.@i]) + " (" + ($@MobDrop_rate[.@i]/100) + "." + ($@MobDrop_rate[.@i]%100) + "%)";
						}
					} else {
						mes "No drops found for this monster.";
					}
					close;
				case 3:
					close;
			}
			break;
		case 3:
			mes "[Market Pulse App]";
			mes "Global Equity Index Trends:";
			query_sql "SELECT mval FROM `solo_stock_meta` WHERE mkey = 'MarketMood'", $MarketMood;
			mes "Market Sentiment: ^0000FF" + callfunc("F_MoodNameStr") + "^000000";
			
			// Latest Breaking News
			set .@ev_hl$, ""; set .@ev_name$, "";
			query_sql "SELECT event_name, headline FROM `solo_stock_events_log` ORDER BY log_id DESC LIMIT 1", .@ev_name$, .@ev_hl$;
			if (.@ev_name$ != "") {
				mes "Latest Headline: ^FF8800" + .@ev_name$ + "^000000";
				mes "  ^777777\"" + .@ev_hl$ + "\"^000000";
			}
			next;
			
			deletearray .@tab_tickers$[0], getarraysize(.@tab_tickers$);
			deletearray .@tab_prices[0], getarraysize(.@tab_prices);
			deletearray .@tab_price_olds[0], getarraysize(.@tab_price_olds);
			.@tab_stock_cnt = query_sql("SELECT ticker, price, price_old FROM `solo_stock_market` WHERE enabled = 1 ORDER BY ticker ASC", .@tab_tickers$, .@tab_prices, .@tab_price_olds);
			
			.@tab_per_page = 5;
			.@tab_total_pages = (.@tab_stock_cnt + .@tab_per_page - 1) / .@tab_per_page;
			for (.@p = 0; .@p < .@tab_total_pages; .@p++) {
				clear;
				mes "[Market Pulse - Page " + (.@p + 1) + "/" + .@tab_total_pages + "]";
				mes "---------------------------";
				.@start_idx = .@p * .@tab_per_page;
				.@end_idx = .@start_idx + .@tab_per_page;
				if (.@end_idx > .@tab_stock_cnt) .@end_idx = .@tab_stock_cnt;
				for (.@i = .@start_idx; .@i < .@end_idx; .@i++) {
					.@price = .@tab_prices[.@i];
					.@price_old = .@tab_price_olds[.@i];
					.@diff = .@price - .@price_old;
					if (.@diff > 0) { .@clr$ = "^00FF00"; .@d$ = "(+" + .@diff + "z)"; }
					else if (.@diff < 0) { .@clr$ = "^FF0000"; .@d$ = "(" + .@diff + "z)"; }
					else { .@clr$ = "^000000"; .@d$ = "(0z)"; }
					mes "- " + .@tab_tickers$[.@i] + ": " + .@clr$ + .@price + "z^000000 " + .@d$;
				}
				next;
			}
			close;
		case 4:
			mes "[Solo Expedition Logistics]";
			query_sql "SELECT `value` FROM `solo_server_config` WHERE `key` = 'expedition_cap_hours' LIMIT 1", .@cap_hours;
			if (.@cap_hours <= 0) {
				query_sql "SELECT `mval` FROM `solo_stock_meta` WHERE `mkey` = 'expedition_cap_hours' LIMIT 1", .@cap_hours;
			}
			if (.@cap_hours <= 0) .@cap_hours = 48;
			.@cap_min = .@cap_hours * 60;
			
			.@available_min = UnclaimedRestMin;
			if (LastLogoutTime > 0) {
				.@available_min += (gettimetick(2) - LastLogoutTime) / 60;
			}
			if (.@available_min > .@cap_min) .@available_min = .@cap_min;
			
			mes "Offline Rest & Expedition time accrues automatically whenever you are logged off.";
			mes "Accumulated Time: ^0000FF" + (.@available_min / 60) + "h " + (.@available_min % 60) + "m^000000 (" + .@cap_hours + "h Cap)";
			
			if (.@available_min < 1) {
				mes " ";
				mes "^FF8800Notice: Minimum 1 minute required to claim rewards.^000000";
				close;
			}
			
			// Calculate estimated rewards (Base EXP and Job EXP only)
			.@est_base = BaseLevel * 10 * .@available_min;
			.@est_job = BaseLevel * 5 * .@available_min;
			
			mes " ";
			mes "Accumulated Vault Yield:";
			mes "- Base EXP: +" + .@est_base;
			mes "- Job EXP: +" + .@est_job;
			next;
			
			if (select("Claim Accumulated Yield:Cancel") == 1) {
				mes "[Solo Expedition Logistics]";
				mes "Expedition rewards claimed successfully!";
				getexp .@est_base, .@est_job;
				UnclaimedRestMin = 0;
				LastLogoutTime = gettimetick(2);
				close;
			}
			close;
		case 5:
			close;
	}
}
