function	script	System_Tablet	{
	doevent "System_Tablet_NPC::OnUse";
	end;
}

-	script	System_Tablet_NPC	-1,{
OnUse:
	mes "[System Tablet]";
	mes "Welcome to the central system.";
	mes "Please select an app to launch:";
	switch(select("Progression Guide:Monster Intel App:Market Pulse App:Cancel")) {
		case 1:
			mes "[System Tablet]";
			mes "Solo-Centric Global Enhancements:";
			mes "^0000FF1.^000000 Your Storage holds up to 800 slots.";
			mes "^0000FF2.^000000 Card Drops are flat 2x (MvP excluded).";
			mes "^0000FF3.^000000 Kafra Warp fees are slashed by 50%.";
			mes "^0000FF4.^000000 Death penalty is reduced to 0.5%.";
			mes " ";
			mes "Progression Note: Experience rates are ^0000FFtiered^000000:";
			mes "- Lv 1-30: 1x";
			mes "- Lv 31-60: 2x";
			mes "- Lv 61-99: 3x";
			mes "- Lv 100-150: 4x";
			mes "- Lv 151+: 5x";
			close;
		case 2:
			mes "[Monster Intel App]";
			mes "Loading Hunters Guild Database...";
			mes "Please input the exact ^0000FFMonster Name^000000 or ^0000FFID^000000 you wish to research.";
			next;
			input .@mob$;
			
			// Determine if input is ID or Name
			if (atoi(.@mob$) > 0 && getmonsterinfo(atoi(.@mob$), MOB_NAME) != "null") {
				.@mob_id = atoi(.@mob$);
			} else if (getmonsterinfo(.@mob$, MOB_ID) > 0) {
				.@mob_id = getmonsterinfo(.@mob$, MOB_ID);
			} else {
				mes "[Monster Intel App]";
				mes "Error: That monster does not exist in the database.";
				close;
			}
			
			.@mob_lv = getmonsterinfo(.@mob_id, MOB_LV);
			.@price_basic = .@mob_lv * 100;
			.@price_deep = .@mob_lv * 500;
			
			// Fetch Unlock Status (0 = Not Unlocked, 1 = Basic, 2 = Deep)
			.@intel_lv = getd("MobIntel_" + .@mob_id);
			
			mes "[Monster Intel App]";
			mes "Target Acquired: ^0000FF" + getmonsterinfo(.@mob_id, MOB_NAME) + "^000000 (Lv " + .@mob_lv + ")";
			
			if (.@intel_lv == 0) mes "Status: ^FF0000[No Intel Unlocked]^000000";
			else if (.@intel_lv == 1) mes "Status: ^FF8800[Basic Intel Unlocked]^000000";
			else mes "Status: ^00FF00[Deep Intel Unlocked]^000000";
			next;
			
			// Dynamic Menu Array builder
			if (.@intel_lv == 0) {
				.@menu$ = "Purchase Basic Intel ("+.@price_basic+"z):Purchase Direct Full Intel ("+(.@price_basic+.@price_deep)+"z):Cancel";
			} else if (.@intel_lv == 1) {
				.@menu$ = "View Basic Intel:Purchase Deep Intel ("+.@price_deep+"z):Cancel";
			} else {
				.@menu$ = "View Basic Intel:View Deep Intel:Cancel";
			}
			
			switch(select(.@menu$)) {
				case 1:
					if (.@intel_lv == 0) {
						if (Zeny < .@price_basic) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= .@price_basic;
						setd "MobIntel_" + .@mob_id, 1;
						mes "[System Tablet]";
						mes "Basic Intel package permanently unlocked!";
						next;
					}
					mes "[Basic Intel]";
					mes "Name: " + getmonsterinfo(.@mob_id, MOB_NAME);
					mes "HP: " + getmonsterinfo(.@mob_id, MOB_MAXHP);
					mes "Level: " + getmonsterinfo(.@mob_id, MOB_LV);
					mes "Element Lv: " + getmonsterinfo(.@mob_id, MOB_ELEMENTLV) + " (Type " + getmonsterinfo(.@mob_id, MOB_ELEMENT) + ")";
					mes "Size: " + getmonsterinfo(.@mob_id, MOB_SIZE);
					mes "Base EXP: " + getmonsterinfo(.@mob_id, MOB_BASEEXP) + " | Job EXP: " + getmonsterinfo(.@mob_id, MOB_JOBEXP);
					close;
					
				case 2:
					if (.@intel_lv == 0) {
						if (Zeny < (.@price_basic + .@price_deep)) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= (.@price_basic + .@price_deep);
						setd "MobIntel_" + .@mob_id, 2;
						mes "[System Tablet]";
						mes "Deep Intel completely unlocked!";
						next;
					} else if (.@intel_lv == 1) {
						if (Zeny < .@price_deep) {
							mes "[System Tablet]";
							mes "Error: Insufficient funds.";
							close;
						}
						Zeny -= .@price_deep;
						setd "MobIntel_" + .@mob_id, 2;
						mes "[System Tablet]";
						mes "Upgraded to Deep Intel package!";
						next;
					}
					
					mes "[Deep Intel]";
					mes "ATK: " + getmonsterinfo(.@mob_id, MOB_ATKMIN) + " ~ " + getmonsterinfo(.@mob_id, MOB_ATKMAX);
					mes "DEF: " + getmonsterinfo(.@mob_id, MOB_DEF) + " | MDEF: " + getmonsterinfo(.@mob_id, MOB_MDEF);
					mes "--- Drops ---";
					
					// Clear previous results to avoid leakage
					deletearray $@MobDrop_item[0], getarraysize($@MobDrop_item);
					deletearray $@MobDrop_rate[0], getarraysize($@MobDrop_rate);
					
					if (getmobdrops(.@mob_id)) {
						.@count = getarraysize($@MobDrop_item);
						for (.@i = 0; .@i < .@count; .@i++) {
							mes "- " + getitemname($@MobDrop_item[.@i]) + " (" + ($@MobDrop_rate[.@i]/100) + "." + ($@MobDrop_rate[.@i]%100) + "%)";
						}
					} else {
						mes "No drops found for this monster.";
					}
					close;
				case 3:
					close;
			}
			break;
		case 3:
			mes "[Market Pulse App]";
			mes "Global Equity Index Trends:";
			mes "Market Sentiment: ^0000FF" + callfunc("F_MoodNameStr") + "^000000";
			mes "---------------------------";
			setarray .@c$[0],"PRT","GEF","MOR","PAY","ALB";
			for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
				set .@cn$, .@c$[.@i]; set .@p, getd("$Price_"+.@cn$); set .@diff, .@p - getd("$Price_"+.@cn$+"_Old");
				if (.@diff > 0) { set .@clr$, "^00FF00"; set .@d$, "(+"+.@diff+"z)"; }
				else if (.@diff < 0) { set .@clr$, "^FF0000"; set .@d$, "("+.@diff+"z)"; }
				else { set .@clr$, "^000000"; set .@d$, "(0z)"; }
				mes "- " + .@cn$ + ": " + .@clr$ + .@p + "z^000000 " + .@d$;
			}
			close;
		case 4:
			close;
	}
}
