// --- Global Logic & Rotation Controller ---
-	script	JunkTraderV2Controller	-1,{
OnInit:
	// Force initial rotation on server boot if uninitialized or new day
	if ($JunkRosterDay != gettime(DT_DAYOFYEAR) || $JunkT1_1 == 0) {
		donpcevent "JunkTraderV2Controller::OnRotateDaily";
	}
	end;

OnClock0000:
OnRotateDaily:
	$JunkRosterDay = gettime(DT_DAYOFYEAR);

	// Dynamically query 3 random items per tier from MySQL table `custom_junk_pool`
	// Tier 1: Novice & 1st Class (Lv 1 - 50)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 1 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT1_1 = .@ids[0]; $JunkPriceT1_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT1_2 = .@ids[1]; $JunkPriceT1_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT1_3 = .@ids[2]; $JunkPriceT1_3 = rand(.@mins[2], .@maxs[2]);
	}

	// Tier 2: 2nd Class & Trans (Lv 51 - 99)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 2 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT2_1 = .@ids[0]; $JunkPriceT2_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT2_2 = .@ids[1]; $JunkPriceT2_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT2_3 = .@ids[2]; $JunkPriceT2_3 = rand(.@mins[2], .@maxs[2]);
	}

	// Tier 3: 3rd Class Early (Lv 100 - 140)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 3 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT3_1 = .@ids[0]; $JunkPriceT3_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT3_2 = .@ids[1]; $JunkPriceT3_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT3_3 = .@ids[2]; $JunkPriceT3_3 = rand(.@mins[2], .@maxs[2]);
	}

	// Tier 4: 3rd Class Mid/High (Lv 141 - 175)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 4 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT4_1 = .@ids[0]; $JunkPriceT4_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT4_2 = .@ids[1]; $JunkPriceT4_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT4_3 = .@ids[2]; $JunkPriceT4_3 = rand(.@mins[2], .@maxs[2]);
	}

	// Tier 5: 3rd Class End & Illusion (Lv 176 - 200)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 5 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT5_1 = .@ids[0]; $JunkPriceT5_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT5_2 = .@ids[1]; $JunkPriceT5_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT5_3 = .@ids[2]; $JunkPriceT5_3 = rand(.@mins[2], .@maxs[2]);
	}

	// Tier 6: 4th Class Range (Lv 201 - 250)
	deletearray .@ids; deletearray .@mins; deletearray .@maxs;
	query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = 6 ORDER BY RAND() LIMIT 3", .@ids, .@mins, .@maxs);
	if (getarraysize(.@ids) >= 3) {
		$JunkT6_1 = .@ids[0]; $JunkPriceT6_1 = rand(.@mins[0], .@maxs[0]);
		$JunkT6_2 = .@ids[1]; $JunkPriceT6_2 = rand(.@mins[1], .@maxs[1]);
		$JunkT6_3 = .@ids[2]; $JunkPriceT6_3 = rand(.@mins[2], .@maxs[2]);
	}

	announce "MARKET NEWS: Daily Junk Merchants have updated their 6-Tier bounty roster!", bc_all;
	end;
}

// --- Main Merchant Function ---
function	script	F_JunkTraderV2Main	{
	// Daily quota tracking
	if (gettime(DT_DAYOFYEAR) != #LastJunkDay) {
		#DailyJunkSold = 0;
		#LastJunkDay = gettime(DT_DAYOFYEAR);
	}

	.@daily_limit = 100;

	mes "[Daily Junk Merchant]";
	mes "Greetings, adventurer! We are purchasing high-value monster scrap today.";
	mes "Prices scale dynamically across all progression tiers.";
	next;

	// Calculate player's level tier
	.@rec_tier = 1;
	if (BaseLevel >= 201)      .@rec_tier = 6;
	else if (BaseLevel >= 176) .@rec_tier = 5;
	else if (BaseLevel >= 141) .@rec_tier = 4;
	else if (BaseLevel >= 100) .@rec_tier = 3;
	else if (BaseLevel >= 51)  .@rec_tier = 2;

	.@tier_name$ = "Tier " + .@rec_tier;
	switch (.@rec_tier) {
		case 1: .@tier_name$ = "Novice / 1st Class (Lv 1-50)"; break;
		case 2: .@tier_name$ = "2nd Class & Trans (Lv 51-99)"; break;
		case 3: .@tier_name$ = "3rd Class Early (Lv 100-140)"; break;
		case 4: .@tier_name$ = "3rd Class Mid/High (Lv 141-175)"; break;
		case 5: .@tier_name$ = "3rd Class End & Illusion (Lv 176-200)"; break;
		case 6: .@tier_name$ = "4th Class Master (Lv 201-250)"; break;
	}

	mes "[Daily Junk Merchant]";
	mes "Your Level: ^0000FF" + BaseLevel + "^000000 (" + .@tier_name$ + ")";
	mes "Today's Quota: ^00FF00" + #DailyJunkSold + " / " + .@daily_limit + " Items^000000";
	next;

	// GM Administration Option
	.@menu$ = "View My Recommended Tier (" + .@tier_name$ + "):Browse All 6 Tiers:Lifetime Trading Stats:Cancel";
	if (getgmlevel() >= 99) {
		.@menu$ = .@menu$ + ":[GM] Admin Controls";
	}

	.@choice = select(.@menu$);

	switch (.@choice) {
		case 1:
			callsub S_TradeTier, .@rec_tier, .@daily_limit;
			break;

		case 2:
			mes "[Daily Junk Merchant]";
			mes "Which level bracket roster would you like to inspect?";
			next;
			.@selected_tier = select("Tier 1: Novice (Lv 1-50):Tier 2: 2nd Class (Lv 51-99):Tier 3: 3rd Class Early (Lv 100-140):Tier 4: 3rd Class High (Lv 141-175):Tier 5: Illusion (Lv 176-200):Tier 6: 4th Class (Lv 201-250):Cancel");
			if (.@selected_tier >= 1 && .@selected_tier <= 6) {
				callsub S_TradeTier, .@selected_tier, .@daily_limit;
			}
			break;

		case 3:
			mes "[Trading Records]";
			mes "Lifetime Items Sold: ^0000FF" + #TotalJunkSold + "^000000";
			mes "Lifetime Zeny Earned: ^00FF00" + #LifetimeZenyEarned + "z^000000";
			close;

		case 5:
			// GM Control Submenu
			mes "[GM Junk Control]";
			mes "Select administrative action:";
			next;
			switch (select("Force Daily Roster Re-roll (from SQL Pool):Reset My Daily Quota:Inspect Pool Counts:Cancel")) {
				case 1:
					donpcevent "JunkTraderV2Controller::OnRotateDaily";
					mes "[GM Junk Control]";
					mes "Daily roster has been re-rolled from SQL pool and broadcasted!";
					close;
				case 2:
					#DailyJunkSold = 0;
					mes "[GM Junk Control]";
					mes "Your daily quota has been reset to 0.";
					close;
				case 3:
					.@count = 0;
					query_sql("SELECT count(*) FROM `custom_junk_pool`", .@count);
					mes "[GM Junk Control]";
					mes "Total items in SQL `custom_junk_pool`: ^0000FF" + .@count + "^000000 items.";
					close;
				default:
					close;
			}
			break;

		default:
			close;
	}
	close;

// --- Subroutine: Trade Specific Tier ---
S_TradeTier:
	.@t = getarg(0);
	.@limit = getarg(1);

	// Load the 3 active items and prices for this tier
	switch (.@t) {
		case 1:
			.@i1 = $JunkT1_1; .@p1 = $JunkPriceT1_1;
			.@i2 = $JunkT1_2; .@p2 = $JunkPriceT1_2;
			.@i3 = $JunkT1_3; .@p3 = $JunkPriceT1_3;
			break;
		case 2:
			.@i1 = $JunkT2_1; .@p1 = $JunkPriceT2_1;
			.@i2 = $JunkT2_2; .@p2 = $JunkPriceT2_2;
			.@i3 = $JunkT2_3; .@p3 = $JunkPriceT2_3;
			break;
		case 3:
			.@i1 = $JunkT3_1; .@p1 = $JunkPriceT3_1;
			.@i2 = $JunkT3_2; .@p2 = $JunkPriceT3_2;
			.@i3 = $JunkT3_3; .@p3 = $JunkPriceT3_3;
			break;
		case 4:
			.@i1 = $JunkT4_1; .@p1 = $JunkPriceT4_1;
			.@i2 = $JunkT4_2; .@p2 = $JunkPriceT4_2;
			.@i3 = $JunkT4_3; .@p3 = $JunkPriceT4_3;
			break;
		case 5:
			.@i1 = $JunkT5_1; .@p1 = $JunkPriceT5_1;
			.@i2 = $JunkT5_2; .@p2 = $JunkPriceT5_2;
			.@i3 = $JunkT5_3; .@p3 = $JunkPriceT5_3;
			break;
		case 6:
			.@i1 = $JunkT6_1; .@p1 = $JunkPriceT6_1;
			.@i2 = $JunkT6_2; .@p2 = $JunkPriceT6_2;
			.@i3 = $JunkT6_3; .@p3 = $JunkPriceT6_3;
			break;
		default:
			return;
	}

	if (.@i1 == 0) {
		mes "[Daily Junk Merchant]";
		mes "The bounty board is currently empty. Please wait for the daily rotation!";
		close;
	}

	mes "[Daily Junk Merchant]";
	mes "Today's Active Bounties for Tier " + .@t + ":";
	mes "1. ^0000FF" + getitemname(.@i1) + "^000000: ^00FF00" + .@p1 + "z^000000";
	mes "2. ^0000FF" + getitemname(.@i2) + "^000000: ^00FF00" + .@p2 + "z^000000";
	mes "3. ^0000FF" + getitemname(.@i3) + "^000000: ^00FF00" + .@p3 + "z^000000";
	next;

	if (#DailyJunkSold >= .@limit) {
		mes "[Daily Junk Merchant]";
		mes "You have reached your daily trade limit (" + .@limit + " items). Please return tomorrow!";
		close;
	}

	.@sell_menu$ = "Sell " + getitemname(.@i1) + " (" + countitem(.@i1) + " in bag):"
	             + "Sell " + getitemname(.@i2) + " (" + countitem(.@i2) + " in bag):"
	             + "Sell " + getitemname(.@i3) + " (" + countitem(.@i3) + " in bag):Cancel";

	.@sell_choice = select(.@sell_menu$);

	switch (.@sell_choice) {
		case 1: .@target_id = .@i1; .@target_price = .@p1; break;
		case 2: .@target_id = .@i2; .@target_price = .@p2; break;
		case 3: .@target_id = .@i3; .@target_price = .@p3; break;
		default: return;
	}

	.@in_bag = countitem(.@target_id);
	if (.@in_bag <= 0) {
		mes "[Daily Junk Merchant]";
		mes "You do not have any ^0000FF" + getitemname(.@target_id) + "^000000 in your inventory.";
		close;
	}

	.@remaining_quota = .@limit - #DailyJunkSold;
	mes "[Daily Junk Merchant]";
	mes "Item: ^0000FF" + getitemname(.@target_id) + "^000000";
	mes "Price: ^00FF00" + .@target_price + "z each^000000";
	mes "You have: ^0000FF" + .@in_bag + "^000000 (Remaining Quota: ^00FF00" + .@remaining_quota + "^000000)";
	mes "How many would you like to sell?";
	next;

	input .@qty;
	if (.@qty <= 0) {
		mes "[Daily Junk Merchant]";
		mes "Transaction cancelled.";
		close;
	}

	if (.@qty > .@remaining_quota) {
		mes "[Daily Junk Merchant]";
		mes "You can only sell up to ^0000FF" + .@remaining_quota + "^000000 more items today.";
		close;
	}

	if (.@qty > .@in_bag) {
		mes "[Daily Junk Merchant]";
		mes "You don't have that many ^0000FF" + getitemname(.@target_id) + "^000000!";
		close;
	}

	delitem .@target_id, .@qty;
	.@payout = .@target_price * .@qty;
	Zeny += .@payout;
	#DailyJunkSold += .@qty;
	#TotalJunkSold += .@qty;
	#LifetimeZenyEarned += .@payout;

	// Record economic metric
	callfunc "F_RecordEconomy", 1, .@qty;
	callfunc "F_RecordEconomy", 2, .@payout;

	mes "[Daily Junk Merchant]";
	mes "Thank you for the business!";
	mes "Sold: ^0000FF" + .@qty + "x " + getitemname(.@target_id) + "^000000";
	mes "Payout: ^00FF00+" + .@payout + "z^000000";
	mes "Remaining Daily Limit: ^0000FF" + (.@limit - #DailyJunkSold) + "^000000";
	close;
}
