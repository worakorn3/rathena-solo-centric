// --- Global Logic & Rotation Controller ---
-	script	JunkTraderV2Controller	-1,{
OnInit:
	// Force initial rotation on server boot if uninitialized or new day
	if ($JunkRosterDay != gettime(DT_DAYOFYEAR) || $JunkT1_1 == 0) {
		donpcevent "JunkTraderV2Controller::OnRotateDaily";
	}
	end;

OnClock0000:
OnRotateDaily:
	$JunkRosterDay = gettime(DT_DAYOFYEAR);
	// Dynamically configure how many items per tier from DB (default to 5)
	$JunkRosterSize = 5;
	query_sql("SELECT `value` FROM `solo_server_config` WHERE `key` = 'junk_roster_size'", .@tmp_size);
	if (.@tmp_size > 0) {
		$JunkRosterSize = .@tmp_size;
	} else {
		// Insert default if missing
		query_sql "INSERT IGNORE INTO `solo_server_config` (`key`, `value`) VALUES ('junk_roster_size', 5)";
	}

	for (.@tier = 1; .@tier <= 6; .@tier++) {
		deletearray .@ids; deletearray .@mins; deletearray .@maxs;
		query_sql("SELECT `item_id`, `min_price`, `max_price` FROM `custom_junk_pool` WHERE `tier` = " + .@tier + " ORDER BY RAND() LIMIT " + $JunkRosterSize, .@ids, .@mins, .@maxs);
		
		for (.@i = 0; .@i < $JunkRosterSize; .@i++) {
			if (.@i < getarraysize(.@ids)) {
				setd "$JunkT" + .@tier + "_" + (.@i + 1), .@ids[.@i];
				setd "$JunkPriceT" + .@tier + "_" + (.@i + 1), rand(.@mins[.@i], .@maxs[.@i]);
			} else {
				setd "$JunkT" + .@tier + "_" + (.@i + 1), 0;
				setd "$JunkPriceT" + .@tier + "_" + (.@i + 1), 0;
			}
		}
	}

	announce "MARKET NEWS: Daily Junk Merchants have updated their 6-Tier bounty roster!", bc_all;
	end;
}

// --- Main Merchant Function ---
function	script	F_JunkTraderV2Main	{
	// Daily quota tracking
	if (gettime(DT_DAYOFYEAR) != #LastJunkDay) {
		#DailyJunkSold = 0;
		#LastJunkDay = gettime(DT_DAYOFYEAR);
	}

	.@daily_limit = 100;

	mes "[Daily Junk Merchant]";
	mes "Greetings, adventurer! We are purchasing high-value monster scrap today.";
	mes "Prices scale dynamically across all progression tiers.";
	next;

	// Calculate player's level tier
	.@rec_tier = 1;
	if (BaseLevel >= 191)      .@rec_tier = 6;
	else if (BaseLevel >= 151) .@rec_tier = 5;
	else if (BaseLevel >= 111) .@rec_tier = 4;
	else if (BaseLevel >= 71)  .@rec_tier = 3;
	else if (BaseLevel >= 31)  .@rec_tier = 2;

	.@tier_name$ = "Tier " + .@rec_tier;
	switch (.@rec_tier) {
		case 1: .@tier_name$ = "Novice (Lv 1-30)"; break;
		case 2: .@tier_name$ = "1st Class (Lv 31-70)"; break;
		case 3: .@tier_name$ = "2nd Class (Lv 71-110)"; break;
		case 4: .@tier_name$ = "Trans/3rd Early (Lv 111-150)"; break;
		case 5: .@tier_name$ = "3rd Class Mid (Lv 151-190)"; break;
		case 6: .@tier_name$ = "4th Class (Lv 191-250)"; break;
	}

	mes "[Daily Junk Merchant]";
	mes "Your Level: ^0000FF" + BaseLevel + "^000000 (" + .@tier_name$ + ")";
	mes "Today's Quota: ^00FF00" + #DailyJunkSold + " / " + .@daily_limit + " Items^000000";
	next;

	// GM Administration Option
	.@menu$ = "View My Recommended Tier (" + .@tier_name$ + "):Browse All 6 Tiers:Lifetime Trading Stats:Cancel";
	if (getgmlevel() >= 99) {
		.@menu$ = .@menu$ + ":[GM] Admin Controls";
	}

	.@choice = select(.@menu$);

	switch (.@choice) {
		case 1:
			callsub S_TradeTier, .@rec_tier, .@daily_limit;
			break;

		case 2:
			mes "[Daily Junk Merchant]";
			mes "Which level bracket roster would you like to inspect?";
			next;
			// ponytail: fixed menu colons to avoid splitting tiers into multiple choices
			.@selected_tier = select("Tier 1 (Lv 1-30):Tier 2 (Lv 31-70):Tier 3 (Lv 71-110):Tier 4 (Lv 111-150):Tier 5 (Lv 151-190):Tier 6 (Lv 191-250):Cancel");
			if (.@selected_tier >= 1 && .@selected_tier <= 6) {
				callsub S_TradeTier, .@selected_tier, .@daily_limit;
			}
			break;

		case 3:
			mes "[Trading Records]";
			mes "Lifetime Items Sold: ^0000FF" + #TotalJunkSold + "^000000";
			mes "Lifetime Zeny Earned: ^00FF00" + #LifetimeZenyEarned + "z^000000";
			close;

		case 5:
			// GM Control Submenu
			mes "[GM Junk Control]";
			mes "Select administrative action:";
			next;
			switch (select("Force Daily Roster Re-roll (from SQL Pool):Reset My Daily Quota:Inspect Pool Counts:Cancel")) {
				case 1:
					donpcevent "JunkTraderV2Controller::OnRotateDaily";
					mes "[GM Junk Control]";
					mes "Daily roster has been re-rolled from SQL pool and broadcasted!";
					close;
				case 2:
					#DailyJunkSold = 0;
					mes "[GM Junk Control]";
					mes "Your daily quota has been reset to 0.";
					close;
				case 3:
					.@count = 0;
					query_sql("SELECT count(*) FROM `custom_junk_pool`", .@count);
					mes "[GM Junk Control]";
					mes "Total items in SQL `custom_junk_pool`: ^0000FF" + .@count + "^000000 items.";
					close;
				default:
					close;
			}
			break;

		default:
			close;
	}
	close;

// --- Subroutine: Trade Specific Tier ---
S_TradeTier:
	.@t = getarg(0);
	.@limit = getarg(1);

	// Load the first item to verify roster isn't empty
	.@first_item = getd("$JunkT" + .@t + "_1");
	if (.@first_item == 0) {
		mes "[Daily Junk Merchant]";
		mes "The bounty board is currently empty. Please wait for the daily rotation!";
		close;
	}

	mes "[Daily Junk Merchant]";
	mes "Today's Active Bounties for Tier " + .@t + ":";
	
	.@sell_menu$ = "";
	for (.@i = 1; .@i <= $JunkRosterSize; .@i++) {
		.@id = getd("$JunkT" + .@t + "_" + .@i);
		.@price = getd("$JunkPriceT" + .@t + "_" + .@i);
		if (.@id > 0) {
			.@in_bag = countitem(.@id);
			.@in_stor = storagecountitem(.@id);
			if (.@in_stor < 0) .@in_stor = 0;
			mes .@i + ". ^0000FF" + getitemname(.@id) + "^000000: ^00FF00" + .@price + "z^000000 [Bag: " + .@in_bag + " | Storage: " + .@in_stor + "]";
			.@sell_menu$ = .@sell_menu$ + "Sell " + getitemname(.@id) + " (Bag " + .@in_bag + " - Storage " + .@in_stor + "):";
			if (.@i % 3 == 0 && .@i < $JunkRosterSize) {
				next;
				mes "[Daily Junk Merchant]";
				mes "Active Bounties for Tier " + .@t + " (Cont.):";
			}
		}
	}
	next;

	if (#DailyJunkSold >= .@limit) {
		mes "[Daily Junk Merchant]";
		mes "You have reached your daily trade limit (" + .@limit + " items). Please return tomorrow!";
		close;
	}

	.@sell_menu$ = .@sell_menu$ + "Cancel";
	.@sell_choice = select(.@sell_menu$);
	
	if (.@sell_choice > $JunkRosterSize || getd("$JunkT" + .@t + "_" + .@sell_choice) == 0) {
		return;
	}

	.@target_id = getd("$JunkT" + .@t + "_" + .@sell_choice);
	.@target_price = getd("$JunkPriceT" + .@t + "_" + .@sell_choice);

	.@in_bag = countitem(.@target_id);
	.@in_stor = storagecountitem(.@target_id);
	if (.@in_stor < 0) .@in_stor = 0;
	.@total_avail = .@in_bag + .@in_stor;

	if (.@total_avail <= 0) {
		mes "[Daily Junk Merchant]";
		mes "You do not have any ^0000FF" + getitemname(.@target_id) + "^000000 in your inventory or storage.";
		close;
	}

	.@remaining_quota = .@limit - #DailyJunkSold;
	mes "[Daily Junk Merchant]";
	mes "Item: ^0000FF" + getitemname(.@target_id) + "^000000 (" + .@target_price + "z each)";
	mes "Inventory: ^0000FF" + .@in_bag + "^000000 | Storage: ^0000FF" + .@in_stor + "^000000";
	mes "Remaining Quota: ^00FF00" + .@remaining_quota + " items^000000";
	mes "How many would you like to sell?";
	next;

	input .@qty;
	if (.@qty <= 0) {
		mes "[Daily Junk Merchant]";
		mes "Transaction cancelled.";
		close;
	}

	if (.@qty > .@remaining_quota) {
		mes "[Daily Junk Merchant]";
		mes "You can only sell up to ^0000FF" + .@remaining_quota + "^000000 more items today.";
		close;
	}

	if (.@qty > .@total_avail) {
		mes "[Daily Junk Merchant]";
		mes "You don't have that many ^0000FF" + getitemname(.@target_id) + "^000000 in bag or storage!";
		close;
	}

	.@from_bag = min(.@qty, .@in_bag);
	.@from_stor = .@qty - .@from_bag;

	if (.@from_bag > 0) {
		delitem .@target_id, .@from_bag;
	}
	if (.@from_stor > 0) {
		storagedelitem .@target_id, .@from_stor;
	}

	.@payout = .@target_price * .@qty;
	Zeny += .@payout;
	#DailyJunkSold += .@qty;
	#TotalJunkSold += .@qty;
	#LifetimeZenyEarned += .@payout;

	// Record economic metric
	callfunc "F_RecordEconomy", 1, .@qty;
	callfunc "F_RecordEconomy", 2, .@payout;

	mes "[Daily Junk Merchant]";
	mes "Thank you for the business!";
	mes "Sold: ^0000FF" + .@qty + "x " + getitemname(.@target_id) + "^000000";
	if (.@from_stor > 0) {
		mes "Source: ^0000FF" + .@from_bag + " Bag^000000, ^0000FF" + .@from_stor + " Storage^000000";
	}
	mes "Payout: ^00FF00+" + .@payout + "z^000000 (Remaining: ^0000FF" + (.@limit - #DailyJunkSold) + "^000000)";
	close;
}
